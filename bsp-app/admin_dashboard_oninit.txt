GV_DATE_FROM = request->get_form_field( 'date_from' ).
GV_DATE_TO   = request->get_form_field( 'date_to' ).

" Default date range: first day of current month to today
IF GV_DATE_FROM IS INITIAL.
  " Default: 4 months ago
  DATA: lv_yr TYPE n LENGTH 4, lv_mo TYPE n LENGTH 2, lv_yy TYPE n LENGTH 4, lv_mm TYPE n LENGTH 2.
  lv_yy = sy-datum(4). lv_mm = sy-datum+4(2).
  lv_yr = lv_yy. lv_mo = lv_mm.
  IF lv_mo > 4.
    lv_mo = lv_mo - 4.
  ELSE.
    lv_yr = lv_yr - 1. lv_mo = lv_mo + 8.
  ENDIF.
  CONCATENATE lv_yr lv_mo '01' INTO GV_DATE_FROM.
ENDIF.
IF GV_DATE_TO IS INITIAL.
  GV_DATE_TO = sy-datum.
ENDIF.

" Convert display dates (YYYY-MM-DD) to SAP internal format (YYYYMMDD)
" Strip dashes first (HTML date input sends YYYY-MM-DD)
REPLACE ALL OCCURRENCES OF '-' IN GV_DATE_FROM WITH ''.
REPLACE ALL OCCURRENCES OF '-' IN GV_DATE_TO WITH ''.

DATA: lv_df_internal TYPE sy-datum,
      lv_dt_internal TYPE sy-datum.
lv_df_internal = GV_DATE_FROM.
lv_dt_internal = GV_DATE_TO.

IF GV_USER_ID IS INITIAL.
  navigation->goto_page( 'login.htm' ).
  RETURN.
ENDIF.

" ================================================================
" 1. Status counts (cards)
" ================================================================
DATA: lt_status TYPE tt_status_count,
      ls_status TYPE ty_status_count.

" Filter: only current month

SELECT status, COUNT(*) AS cnt
  INTO TABLE @lt_status
  FROM zecm_req
  WHERE created_date >= @lv_df_internal AND created_date <= @lv_dt_internal
  GROUP BY status.

LOOP AT lt_status INTO ls_status.
  CASE ls_status-status.
    WHEN 'CRTD'. GV_COUNT_CRTD = ls_status-count.
    WHEN 'INWK'. GV_COUNT_INWK = ls_status-count.
    WHEN 'BMPR'. GV_COUNT_BMPR = ls_status-count.
    WHEN 'BMAP'. GV_COUNT_BMAP = ls_status-count.
    WHEN 'INSP'. GV_COUNT_INSP = ls_status-count.
    WHEN 'REJC'. GV_COUNT_REJC = ls_status-count.
  ENDCASE.
ENDLOOP.

" ================================================================
" 2. Monthly aggregation (last 6 months)
" ================================================================
DATA: lt_monthly TYPE tt_monthly,
      ls_monthly TYPE ty_monthly.

* Monthly aggregation filtered by date range
DATA: BEGIN OF ls_raw,
        created_date TYPE zecm_req-created_date,
        status       TYPE zecm_req-status,
      END OF ls_raw,
      lt_raw LIKE STANDARD TABLE OF ls_raw.

SELECT created_date, status
  INTO TABLE @lt_raw
  FROM zecm_req
  WHERE created_date >= @lv_df_internal AND created_date <= @lv_dt_internal.

" --- ADD IREQ (Request Masuk) TO RAW DATA ---
DATA: BEGIN OF ls_raw_ireq,
        req_date TYPE zecm_ireq-req_date,
        status   TYPE zecm_ireq-status,
      END OF ls_raw_ireq,
      lt_raw_ireq LIKE STANDARD TABLE OF ls_raw_ireq.

SELECT req_date, status
  INTO TABLE @lt_raw_ireq
  FROM zecm_ireq
  WHERE req_date >= @lv_df_internal AND req_date <= @lv_dt_internal
    AND status = 'REQ'.

LOOP AT lt_raw_ireq INTO ls_raw_ireq.
  ls_raw-created_date = ls_raw_ireq-req_date.
  ls_raw-status       = 'IREQ'. " Fake status to group Request Masuk
  APPEND ls_raw TO lt_raw.
ENDLOOP.
" ------------------------------------------

DATA: ls_agg TYPE ty_monthly.
CLEAR lt_monthly.

LOOP AT lt_raw INTO ls_raw.
  ls_agg-month  = ls_raw-created_date(6).
  ls_agg-status = ls_raw-status.
  READ TABLE lt_monthly INTO ls_agg
    WITH KEY month = ls_agg-month status = ls_agg-status.
  IF sy-subrc = 0.
    ls_agg-count = ls_agg-count + 1.
    MODIFY lt_monthly FROM ls_agg INDEX sy-tabix.
  ELSE.
    ls_agg-count = 1.
    APPEND ls_agg TO lt_monthly.
  ENDIF.
ENDLOOP.

" Build JSON arrays for Chart.js
DATA: lv_month_label TYPE string,
      lv_prev_month  TYPE c LENGTH 6 VALUE ''.

DATA: lv_m_ireq TYPE i,

    FROM zecm_req AS a

    LEFT OUTER JOIN zecm_user AS c
      ON a~requester = c~zecm_user_id
    LEFT OUTER JOIN zecm_user AS d
      ON a~drafter = d~zecm_user_id
    WHERE ( a~matnr_fg LIKE @lv_kw OR b~maktx LIKE @lv_kw )
    ORDER BY a~created_date DESCENDING, a~created_time DESCENDING.

  " Filter out routing requests from material history manually
  DELETE lt_mat_hist WHERE change_scope = 'ROUT' OR title CS 'Routing' OR title CS 'alur proses'.

  " Build HTML cards
  DATA: lv_s_matnr   TYPE string,
        lv_s_ecr     TYPE string,
        lv_s_status  TYPE string,
        lv_s_badge   TYPE string,

      READ TABLE lv_dir_parts_m INTO lv_dir_doknr_m INDEX 2.

    ENDCASE.

    " Priority color
    CASE ls_mat_hist-priority.
      WHEN 'CRITICAL'. lv_s_pri_clr = 'bg-red-100 text-red-700'.
      WHEN 'HIGH'.     lv_s_pri_clr = 'bg-orange-100 text-orange-700'.
      WHEN 'MEDIUM'.   lv_s_pri_clr = 'bg-blue-100 text-blue-700'.
      WHEN OTHERS.     lv_s_pri_clr = 'bg-slate-100 text-slate-600'.
    ENDCASE.

    " Kategori color

DATA: lt_ireq          TYPE TABLE OF zecm_ireq,
      ls_ireq          TYPE zecm_ireq,
      lv_ireq_reqname  TYPE string,
      lv_ireq_row      TYPE string.

SELECT * FROM zecm_ireq INTO TABLE @lt_ireq
  WHERE status = 'APPR' OR status = 'REQ'
  ORDER BY req_date DESCENDING.

" Simpan nilai sy-subrc dari SELECT asli sebelum ditimpa oleh perintah DELETE ABAP
DATA: lv_subrc_ireq TYPE sy-subrc.
lv_subrc_ireq = sy-subrc.

" Filter out routing requests from IREQ panel manually
DELETE lt_ireq WHERE title CS 'Routing' OR title CS 'alur proses' OR title CS 'routing' OR title CS 'ROUTING'.

" Sort lt_ireq by status first (REQ on top, APPR next), then by date descending
" Since REQ > APPR alphabetically, descending sort by status works perfectly
SORT lt_ireq BY status DESCENDING req_date DESCENDING.

IF lv_subrc_ireq = 0 AND lt_ireq IS NOT INITIAL.
  GV_COUNT_IREQ = lines( lt_ireq ).

  LOOP AT lt_ireq INTO ls_ireq.
    CLEAR: lv_ireq_row, lv_ireq_reqname.

    " Get requester name
    SELECT SINGLE zecm_full_name FROM zecm_user
      INTO @lv_ireq_reqname
      WHERE zecm_user_id = @ls_ireq-requester.

    " Bersihkan leading zero material & ambil deskripsi
    DATA: lv_ireq_mat_cln TYPE string,
          lv_ireq_mat_desc TYPE string,
          lv_kategori_clr  TYPE string,
          lv_priority_clr  TYPE string.
          
    lv_ireq_mat_cln = ls_ireq-matnr_fg.
    SHIFT lv_ireq_mat_cln LEFT DELETING LEADING '0'.
    
    " Ambil deskripsi material dari tabel MAKT
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING input = ls_ireq-matnr_fg
      IMPORTING output = lv_fix_matnr.
    SELECT SINGLE maktx FROM makt INTO @lv_ireq_mat_desc
      WHERE matnr = @lv_fix_matnr AND spras = @sy-langu.
    
    " Styling Kategori
    CASE ls_ireq-kategori.
      WHEN 'DRW'. lv_kategori_clr = 'bg-emerald-100 text-emerald-700 border-emerald-200'.
      WHEN 'DBX'. lv_kategori_clr = 'bg-blue-100 text-blue-700 border-blue-200'.
      WHEN 'DMT'. lv_kategori_clr = 'bg-orange-100 text-orange-700 border-orange-200'.
      WHEN OTHERS. lv_kategori_clr = 'bg-slate-100 text-slate-600 border-slate-200'.
    ENDCASE.
    
    " Styling Prioritas
    CASE ls_ireq-priority.
      WHEN 'CRITICAL'. lv_priority_clr = 'bg-red-100 text-red-700'.
      WHEN 'HIGH'.     lv_priority_clr = 'bg-orange-100 text-orange-700'.
      WHEN 'MEDIUM'.   lv_priority_clr = 'bg-blue-100 text-blue-700'.
      WHEN OTHERS.     lv_priority_clr = 'bg-slate-100 text-slate-600'.
    ENDCASE.

    " Tanggal formatter
    DATA: lv_ireq_dt_fmt TYPE string.
    lv_ireq_dt_fmt = |{ ls_ireq-req_date+6(2) }.{ ls_ireq-req_date+4(2) }.{ ls_ireq-req_date(4) }|.
    DATA: lv_ireq_stat_clr TYPE string,
          lv_ireq_stat_lbl TYPE string.
    CASE ls_ireq-status.
      WHEN 'REQ'.
        lv_ireq_stat_clr = 'bg-amber-100 text-amber-700'.
        lv_ireq_stat_lbl = 'Pending Review'.
      WHEN 'APPR'.
        lv_ireq_stat_clr = 'bg-blue-100 text-blue-700'.
        lv_ireq_stat_lbl = 'Approved'.
      WHEN 'ECO'.
        lv_ireq_stat_clr = 'bg-emerald-100 text-emerald-700'.
        lv_ireq_stat_lbl = 'ECO Created'.
      WHEN 'REJC'.
        lv_ireq_stat_clr = 'bg-red-100 text-red-700'.
        lv_ireq_stat_lbl = 'Rejected'.
      WHEN OTHERS.
        lv_ireq_stat_clr = 'bg-slate-100 text-slate-600'.
        lv_ireq_stat_lbl = ls_ireq-status.
    ENDCASE.

    " Build table row
    CONCATENATE 
      '<tr class="hover:bg-slate-50">'
      '<td class="px-4 py-3 text-xs font-mono text-brand-600 font-semibold">'
      ls_ireq-ireq_id+0(10) '</td>'
      '<td class="px-4 py-3 text-sm font-semibold text-slate-700 max-w-[200px] truncate" title="' ls_ireq-title '">'
      ls_ireq-title '</td>'
      '<td class="px-4 py-3">'
      '<div class="font-semibold text-sm text-slate-700">' lv_ireq_mat_cln '</div>'
      '<div class="text-xs text-slate-400 max-w-xs truncate" title="' lv_ireq_mat_desc '">' lv_ireq_mat_desc '</div>'
      '</td>'
      '<td class="px-4 py-3"><span class="px-2.5 py-0.5 rounded text-[0.65rem] font-bold border ' lv_kategori_clr '">' ls_ireq-kategori '</span></td>'
      '<td class="px-4 py-3"><span class="px-2.5 py-0.5 rounded-full text-[0.65rem] font-bold ' lv_priority_clr '">' ls_ireq-priority '</span></td>'
      '<td class="px-4 py-3 text-sm text-slate-600">'
      lv_ireq_reqname '</td>'
      '<td class="px-4 py-3 text-xs text-slate-500 font-semibold">'
      lv_ireq_dt_fmt '</td>'
      '<td class="px-4 py-3"><span class="px-2.5 py-0.5 rounded-full text-[0.65rem] font-bold ' lv_ireq_stat_clr '">' lv_ireq_stat_lbl '</span></td>'
      '<td class="px-4 py-3">'
      '<a href="admin_req_review.htm?u='
      INTO lv_ireq_row.

    " V11_i: Selalu arahkan ke admin_req_review, disable-nya ada di dalam halaman detailnya
    CONCATENATE lv_ireq_row
      GV_USER_ID '&r=' GV_ROLE '&n=' GV_FULL_NAME

      '<i class="fas fa-eye mr-1"></i>Review</a></td>'
      '</tr>'
      INTO lv_ireq_row.
    
    CONCATENATE GV_IREQ_HTML lv_ireq_row INTO GV_IREQ_HTML.
  ENDLOOP.
ELSE.
  GV_COUNT_IREQ = 0.
