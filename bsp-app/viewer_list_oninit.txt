GV_DATE_TO   = request->get_form_field( 'date_to' ).

" Default date range: first day of current month to today
IF GV_DATE_FROM IS INITIAL.
  " Default: 4 months ago
  DATA: lv_yr TYPE n LENGTH 4, lv_mo TYPE n LENGTH 2, lv_yy TYPE n LENGTH 4, lv_mm TYPE n LENGTH 2.
  lv_yy = sy-datum(4). lv_mm = sy-datum+4(2).
  lv_yr = lv_yy. lv_mo = lv_mm.
  IF lv_mo > 4.
    lv_mo = lv_mo - 4.
  ELSE.
    lv_yr = lv_yr - 1. lv_mo = lv_mo + 8.
  ENDIF.
  CONCATENATE lv_yr lv_mo '01' INTO GV_DATE_FROM.
ENDIF.
IF GV_DATE_TO IS INITIAL.
  GV_DATE_TO = sy-datum.
ENDIF.

" Strip dashes (HTML date input sends YYYY-MM-DD)
REPLACE ALL OCCURRENCES OF '-' IN GV_DATE_FROM WITH ''.
REPLACE ALL OCCURRENCES OF '-' IN GV_DATE_TO WITH ''.

DATA: lv_df_internal TYPE sy-datum,
      lv_dt_internal TYPE sy-datum.
lv_df_internal = GV_DATE_FROM.
lv_dt_internal = GV_DATE_TO.

  LV_FILTER = 'ALL'.

DATA: lt_monthly TYPE tt_monthly,

DATA: BEGIN OF ls_raw,
        created_date TYPE zecm_req-created_date,
        status       TYPE zecm_req-status,

      lt_raw LIKE STANDARD TABLE OF ls_raw.

SELECT created_date, status
  INTO TABLE @lt_raw
  FROM zecm_req
  WHERE created_date >= @lv_df_internal AND created_date <= @lv_dt_internal.

" Add IREQ (Request Masuk) TO RAW DATA
DATA: BEGIN OF ls_raw_ireq,
        req_date TYPE zecm_ireq-req_date,
        status   TYPE zecm_ireq-status,
      END OF ls_raw_ireq,
      lt_raw_ireq LIKE STANDARD TABLE OF ls_raw_ireq.

SELECT req_date, status
  INTO TABLE @lt_raw_ireq
  FROM zecm_ireq
  WHERE req_date >= @lv_df_internal AND req_date <= @lv_dt_internal
    AND status = 'REQ'.

LOOP AT lt_raw_ireq INTO ls_raw_ireq.
  ls_raw-created_date = ls_raw_ireq-req_date.
  ls_raw-status       = 'IREQ'. " Fake status to group Request Masuk
  APPEND ls_raw TO lt_raw.
ENDLOOP.

DATA: ls_agg TYPE ty_monthly.
CLEAR lt_monthly.

LOOP AT lt_raw INTO ls_raw.
  ls_agg-month  = ls_raw-created_date(6).
  ls_agg-status = ls_raw-status.
  READ TABLE lt_monthly INTO ls_agg
    WITH KEY month = ls_agg-month status = ls_agg-status.
  IF sy-subrc = 0.
    ls_agg-count = ls_agg-count + 1.
    MODIFY lt_monthly FROM ls_agg INDEX sy-tabix.
  ELSE.
    ls_agg-count = 1.
    APPEND ls_agg TO lt_monthly.
  ENDIF.
ENDLOOP.

" Build JSON arrays for Chart.js
DATA: lv_month_label TYPE string,
      lv_prev_month  TYPE c LENGTH 6 VALUE ''.

DATA: lv_m_ireq TYPE i,
      lv_m_crtd TYPE i,
      lv_m_inwk TYPE i,
      lv_m_bmpr TYPE i,
      lv_m_bmap TYPE i,
      lv_m_insp TYPE i,

      lv_m_bmap_s TYPE string,
      lv_m_insp_s TYPE string,
      lv_m_rejc_s TYPE string.

CLEAR: lv_labels, lv_d_crtd, lv_d_inwk, lv_d_insp, lv_d_rejc, lv_comma.

  IF ls_monthly-month <> lv_prev_month.

      lv_m_ireq_s = lv_m_ireq.
      lv_m_crtd_s = lv_m_crtd.
      lv_m_inwk_s = lv_m_inwk.
      lv_m_bmpr_s = lv_m_bmpr.
      lv_m_bmap_s = lv_m_bmap.
      lv_m_insp_s = lv_m_insp.
      lv_m_rejc_s = lv_m_rejc.

    lv_prev_month = ls_monthly-month.

      WHEN '01'. lv_month_label = 'Jan'.

      WHEN '03'. lv_month_label = 'Mar'.
      WHEN '04'. lv_month_label = 'Apr'.
      WHEN '05'. lv_month_label = 'Mei'.
      WHEN '06'. lv_month_label = 'Jun'.

      WHEN '11'. lv_month_label = 'Nov'.
      WHEN '12'. lv_month_label = 'Des'.
    ENDCASE.
  ENDIF.

  CASE ls_monthly-status.
    WHEN 'IREQ'. lv_m_ireq = lv_m_ireq + ls_monthly-count.

    WHEN 'BMAP'. lv_m_bmap = lv_m_bmap + ls_monthly-count.
    WHEN 'INSP'. lv_m_insp = lv_m_insp + ls_monthly-count.
    WHEN 'REJC'. lv_m_rejc = lv_m_rejc + ls_monthly-count.

  lv_m_inwk_s = lv_m_inwk.
  lv_m_bmpr_s = lv_m_bmpr.
  lv_m_bmap_s = lv_m_bmap.
  lv_m_insp_s = lv_m_insp.

IF LV_ECR_DETAIL IS NOT INITIAL.

  SELECT SINGLE a~ecr_no, a~title, a~matnr_fg, a~descr, a~kategori, a~priority,
         a~requester, a~drafter, a~dokar, a~dir_number, a~status,
         a~reject_reason, a~cost_estimate, a~zdept, a~zreason,
         a~created_date, a~changed_date,
         c~zecm_full_name AS requester_name,

         b~maktx
    INTO CORRESPONDING FIELDS OF @LS_ECR_DETAIL
    FROM zecm_req AS a
    LEFT OUTER JOIN makt AS b
      ON a~matnr_fg = b~matnr AND b~spras = @sy-langu
    LEFT OUTER JOIN zecm_user AS c
      ON a~requester = c~zecm_user_id
    LEFT OUTER JOIN zecm_user AS d
      ON a~drafter = d~zecm_user_id
    WHERE a~ecr_no = @LV_ECR_DETAIL.

      IMPORTING  output = lv_matnr_detail.

      EXPORTING  input  = lv_matnr_detail
      IMPORTING  output = lv_matnr_raw_d.

      INTO (@lv_doknr_detail, @lv_dokar_detail, @lv_dokvr_detail, @lv_doktl_detail, @lv_dokst_detail)

      WHERE dokst IN ('IA', 'AP', 'FR')

      SELECT SINGLE objky INTO @lv_objky_detail
        FROM drad
        WHERE dokar = @lv_dokar_detail
          AND doknr = @lv_doknr_detail
          AND dokvr = @lv_dokvr_detail
          AND doktl = @lv_doktl_detail.

      IF sy-subrc = 0.
        CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'

          IMPORTING  output = lv_objky_clean_d.
        IF lv_objky_clean_d = lv_matnr_raw_d.
          LS_ECR_DETAIL-dms_doknr = lv_doknr_detail.
          LS_ECR_DETAIL-dms_dokar = lv_dokar_detail.
          LS_ECR_DETAIL-dms_dokvr = lv_dokvr_detail.
          LS_ECR_DETAIL-dms_doktl = lv_doktl_detail.
          LS_ECR_DETAIL-dms_dokst = lv_dokst_detail.

      LS_ECR_DETAIL-dms_found = abap_true.
      SELECT SINGLE dktxt FROM drat INTO @LS_ECR_DETAIL-dms_dktxt
        WHERE dokar = @LS_ECR_DETAIL-dms_dokar
          AND doknr = @LS_ECR_DETAIL-dms_doknr
          AND dokvr = @LS_ECR_DETAIL-dms_dokvr
          AND doktl = @LS_ECR_DETAIL-dms_doktl.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
        EXPORTING  input  = LS_ECR_DETAIL-dms_doknr
        IMPORTING  output = LS_ECR_DETAIL-dms_doknr.
    ENDIF.

    " Clean dir_number
    IF LS_ECR_DETAIL-dir_number IS NOT INITIAL.
      DATA: lv_dir_parts_d TYPE STANDARD TABLE OF string,
            lv_dir_doknr_d TYPE string.
      SPLIT LS_ECR_DETAIL-dir_number AT '/' INTO TABLE lv_dir_parts_d.
      READ TABLE lv_dir_parts_d INTO lv_dir_doknr_d INDEX 2.
      IF lv_dir_doknr_d IS NOT INITIAL.
        SHIFT lv_dir_doknr_d LEFT DELETING LEADING '0'.
        LS_ECR_DETAIL-dir_number_clean = lv_dir_doknr_d.
      ELSE.
        LS_ECR_DETAIL-dir_number_clean = LS_ECR_DETAIL-dir_number.
      ENDIF.
    ENDIF.
  ENDIF.

  " Audit Trail
  SELECT log_id, ecr_no, action, old_status, new_status,
         user_id, e~zecm_full_name AS user_name, action_date, action_time, reason
         INTO TABLE @LT_LOGS
         FROM zecm_log
         LEFT OUTER JOIN zecm_user AS e
           ON zecm_log~user_id = e~zecm_user_id
         WHERE ecr_no = @LV_ECR_DETAIL
        AND is_ireq = ''
         ORDER BY action_date DESCENDING, action_time DESCENDING.

  LOOP AT LT_LOGS INTO LS_LOG.
    CASE LS_LOG-old_status.
      WHEN 'CRTD'. LS_LOG-old_label = 'Draft'.
      WHEN 'INWK'. LS_LOG-old_label = 'In Work'.
      WHEN 'INSP'. LS_LOG-old_label = 'Inspection'.
      WHEN 'CLSD'. LS_LOG-old_label = 'Closed'.
      WHEN 'REJC'. LS_LOG-old_label = 'Rejected'.
      WHEN OTHERS. LS_LOG-old_label = LS_LOG-old_status.
    ENDCASE.
    CASE LS_LOG-new_status.
      WHEN 'CRTD'. LS_LOG-new_label = 'Draft'.
      WHEN 'INWK'. LS_LOG-new_label = 'In Work'.
      WHEN 'INSP'. LS_LOG-new_label = 'Inspection'.
      WHEN 'CLSD'. LS_LOG-new_label = 'Closed'.
      WHEN 'REJC'. LS_LOG-new_label = 'Rejected'.
      WHEN OTHERS. LS_LOG-new_label = LS_LOG-new_status.
    ENDCASE.
    CASE LS_LOG-action.
      WHEN 'CREA'. LS_LOG-action_label = 'Created'.
      WHEN 'APPR'. LS_LOG-action_label = 'Approved'.
      WHEN 'INSP'. LS_LOG-action_label = 'Submitted'.
      WHEN 'REJT'. LS_LOG-action_label = 'Rejected'.
      WHEN 'REVS'. LS_LOG-action_label = 'Revised'.
      WHEN 'CLSD'. LS_LOG-action_label = 'Closed'.
      WHEN 'CANC'. LS_LOG-action_label = 'Cancelled'.
      WHEN OTHERS. LS_LOG-action_label = LS_LOG-action.
    ENDCASE.
    MODIFY LT_LOGS FROM LS_LOG.
  ENDLOOP.

ELSE.

  " ================================================================
  " 6. ECR List (ALL ECRs for viewer)
  " ================================================================
  " ECR is ALWAYS Draft (CRTD) regardless of filter
  SELECT a~ecr_no, a~title, a~matnr_fg, a~descr, a~kategori, a~priority,
         a~requester, a~drafter, a~dokar, a~dir_number, a~status,
         a~reject_reason, a~cost_estimate, a~zdept, a~zreason,
         a~created_date, a~created_time, a~changed_date,
         c~zecm_full_name AS requester_name,
         d~zecm_full_name AS drafter_name,
         b~maktx
    INTO CORRESPONDING FIELDS OF TABLE @LT_ECR
    FROM zecm_req AS a
    LEFT OUTER JOIN makt AS b
      ON a~matnr_fg = b~matnr AND b~spras = @sy-langu
    LEFT OUTER JOIN zecm_user AS c
      ON a~requester = c~zecm_user_id
    LEFT OUTER JOIN zecm_user AS d
      ON a~drafter = d~zecm_user_id
    WHERE a~status = 'CRTD'
    ORDER BY a~created_date DESCENDING, a~created_time DESCENDING.

  " Process each ECR
  DATA: lv_matnr_conv_a TYPE matnr,
        lv_doknr_a TYPE doknr,
        lv_dokar_a TYPE dokar,
        lv_dokvr_a TYPE char2,
        lv_doktl_a TYPE char3,
        lv_dokst_a TYPE char2,
        lv_objky_a TYPE char90,
        lv_matnr_raw_a TYPE char90,
        lv_objky_clean_a TYPE char90,
        lv_dir_parts TYPE STANDARD TABLE OF string,
        lv_dir_doknr TYPE string.

  DATA: lv_target_dokst TYPE char2.

  LOOP AT LT_ECR INTO LS_ECR.
    " Remove leading zeros from ECR number
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
      EXPORTING  input  = LS_ECR-ecr_no
      IMPORTING  output = LS_ECR-ecr_no.

    " Material no zero
    LS_ECR-matnr_fg_no_zero = LS_ECR-matnr_fg.
    SHIFT LS_ECR-matnr_fg_no_zero LEFT DELETING LEADING '0'.

    " Clean dir_number
    IF LS_ECR-dir_number IS NOT INITIAL.
      SPLIT LS_ECR-dir_number AT '/' INTO TABLE lv_dir_parts.
      READ TABLE lv_dir_parts INTO lv_dir_doknr INDEX 2.
      IF lv_dir_doknr IS NOT INITIAL.
        SHIFT lv_dir_doknr LEFT DELETING LEADING '0'.
        LS_ECR-dir_number_clean = lv_dir_doknr.
      ELSE.
        LS_ECR-dir_number_clean = LS_ECR-dir_number.
      ENDIF.
    ENDIF.

    " Target DMS status based on ECR status
    " Viewer sees all statuses, so check IA, AP, FR
    IF LS_ECR-status = 'INWK' OR LS_ECR-status = 'BMAP'.
      lv_target_dokst = 'IA'.
    ELSEIF LS_ECR-status = 'INSP'.
      lv_target_dokst = 'AP'.
    ELSEIF LS_ECR-status = 'CLSD'.
      lv_target_dokst = 'FR'.
    ELSE.
      CLEAR lv_target_dokst.
    ENDIF.

    " DMS detection — V04 approach
    IF lv_target_dokst IS NOT INITIAL.
      lv_matnr_conv_a = LS_ECR-matnr_fg.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
        EXPORTING  input  = lv_matnr_conv_a
        IMPORTING  output = lv_matnr_conv_a.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
        EXPORTING  input  = lv_matnr_conv_a
        IMPORTING  output = lv_matnr_raw_a.

      CLEAR: lv_doknr_a, lv_dokar_a, lv_dokvr_a, lv_doktl_a, lv_dokst_a.

      SELECT doknr, dokar, dokvr, doktl, dokst
        INTO (@lv_doknr_a, @lv_dokar_a, @lv_dokvr_a, @lv_doktl_a, @lv_dokst_a)
        FROM draw
        WHERE dokst = @lv_target_dokst
          AND dokar IN ('DRW', 'DBX', 'DMT').

        SELECT SINGLE objky INTO @lv_objky_a
          FROM drad
          WHERE dokar = @lv_dokar_a
            AND doknr = @lv_doknr_a
            AND dokvr = @lv_dokvr_a
            AND doktl = @lv_doktl_a.

        IF sy-subrc = 0.
          CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
            EXPORTING  input  = lv_objky_a
            IMPORTING  output = lv_objky_clean_a.
          IF lv_objky_clean_a = lv_matnr_raw_a.
            LS_ECR-dms_doknr = lv_doknr_a.
            LS_ECR-dms_dokar = lv_dokar_a.
            LS_ECR-dms_dokvr = lv_dokvr_a.
            LS_ECR-dms_doktl = lv_doktl_a.
            LS_ECR-dms_dokst = lv_dokst_a.
            EXIT.
          ENDIF.
        ENDIF.
      ENDSELECT.

      IF LS_ECR-dms_doknr IS NOT INITIAL.
        LS_ECR-dms_found = abap_true.
        SELECT SINGLE dktxt FROM drat INTO @LS_ECR-dms_dktxt
          WHERE dokar = @LS_ECR-dms_dokar
            AND doknr = @LS_ECR-dms_doknr
            AND dokvr = @LS_ECR-dms_dokvr
            AND doktl = @LS_ECR-dms_doktl.
        CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
          EXPORTING  input  = LS_ECR-dms_doknr
          IMPORTING  output = LS_ECR-dms_doknr.
      ENDIF.
    ENDIF.

    MODIFY LT_ECR FROM LS_ECR.
  ENDLOOP.

ENDIF.

" ================================================================
" ================================================================
" 7. ECO List (In Work, BOM Pending, BOM Approved, Inspection, Rejected)
" ================================================================
CLEAR: GV_ECO_HTML.
DATA: lt_eco_v       TYPE TABLE OF zecm_req,
      ls_eco_v       TYPE zecm_req,
      lv_eco_name_v  TYPE string,
      lv_sts_color   TYPE string,
      lv_sts_label   TYPE string.

IF LV_FILTER = 'ALL' OR LV_FILTER = 'CRTD'.
  " Show all ECOs if filter is ALL (or if they somehow clicked CRTD)
  SELECT * FROM zecm_req INTO TABLE @lt_eco_v
    WHERE status IN ('INWK', 'BMPR', 'BMAP', 'INSP', 'REJC')
    ORDER BY created_date DESCENDING.
ELSE.
  " Show specific ECO status
  SELECT * FROM zecm_req INTO TABLE @lt_eco_v
    WHERE status = @LV_FILTER
    ORDER BY created_date DESCENDING.
ENDIF.

IF sy-subrc = 0.
  LOOP AT lt_eco_v INTO ls_eco_v.
    SELECT SINGLE zecm_full_name FROM zecm_user
      INTO @lv_eco_name_v
      WHERE zecm_user_id = @ls_eco_v-requester.
    IF sy-subrc <> 0.
      CLEAR lv_eco_name_v.
    ENDIF.

    CLEAR: lv_kat_color, lv_pri_color, lv_sts_color, lv_sts_label.
    " Kategori Color (inline style agar tidak kena Tailwind purge)
    CASE ls_eco_v-kategori.
      WHEN 'DRW'. lv_kat_color = 'background:#DBEAFE;color:#1D4ED8'.
      WHEN 'DBX'. lv_kat_color = 'background:#FFEDD5;color:#C2410C'.
      WHEN 'DMT'. lv_kat_color = 'background:#D1FAE5;color:#065F46'.
      WHEN OTHERS. lv_kat_color = 'background:#F1F5F9;color:#475569'.
    ENDCASE.

    " Priority Color (inline style)
    CASE ls_eco_v-priority.
      WHEN 'CRITICAL'. lv_pri_color = 'background:#FEE2E2;color:#B91C1C'.
      WHEN 'HIGH'.     lv_pri_color = 'background:#FFEDD5;color:#C2410C'.
      WHEN 'MEDIUM'.   lv_pri_color = 'background:#FEF9C3;color:#854D0E'.
      WHEN 'LOW'.      lv_pri_color = 'background:#F1F5F9;color:#475569'.
      WHEN OTHERS.     lv_pri_color = 'background:#F1F5F9;color:#475569'.
    ENDCASE.

    " Status Color (inline style)
    CLEAR lv_sts_label.
    CASE ls_eco_v-status.
      WHEN 'INWK'. lv_sts_color = 'background:#FFEDD5;color:#C2410C'. lv_sts_label = 'In Work'.
      WHEN 'BMPR'. lv_sts_color = 'background:#FEF3C7;color:#B45309'. lv_sts_label = 'BOM Pending'.
      WHEN 'BMAP'. lv_sts_color = 'background:#CCFBF1;color:#0F766E'. lv_sts_label = 'BOM Approved'.
      WHEN 'INSP'. lv_sts_color = 'background:#FEF9C3;color:#854D0E'. lv_sts_label = 'Inspection'.
      WHEN 'REJC'. lv_sts_color = 'background:#FEE2E2;color:#B91C1C'. lv_sts_label = 'Rejected'.
      WHEN OTHERS. lv_sts_color = 'background:#F1F5F9;color:#475569'. lv_sts_label = ls_eco_v-status.
    ENDCASE.

    CONCATENATE GV_ECO_HTML '<tr class="hover:bg-slate-50">' INTO GV_ECO_HTML.
    CONCATENATE GV_ECO_HTML '<td class="px-4 py-3 font-bold text-brand-600 font-mono">' ls_eco_v-ecr_no '</td>' INTO GV_ECO_HTML.
    CONCATENATE GV_ECO_HTML '<td class="px-4 py-3 text-slate-700 max-w-[200px] truncate">' ls_eco_v-title '</td>' INTO GV_ECO_HTML.
    CONCATENATE GV_ECO_HTML '<td class="px-4 py-3 text-xs font-semibold text-slate-700">' ls_eco_v-matnr_fg '</td>' INTO GV_ECO_HTML.
    CONCATENATE GV_ECO_HTML '<td class="px-4 py-3"><span class="px-2 py-1 rounded-md text-[0.65rem] font-bold uppercase tracking-wide" style="' lv_kat_color '">' ls_eco_v-kategori '</span></td>' INTO GV_ECO_HTML.
    CONCATENATE GV_ECO_HTML '<td class="px-4 py-3"><span class="px-2 py-1 rounded-md text-[0.65rem] font-bold uppercase tracking-wide" style="' lv_pri_color '">' ls_eco_v-priority '</span></td>' INTO GV_ECO_HTML.
    CONCATENATE GV_ECO_HTML '<td class="px-4 py-3"><span class="px-2 py-1 rounded-md text-[0.65rem] font-bold uppercase tracking-wide" style="' lv_sts_color '">' lv_sts_label '</span></td>' INTO GV_ECO_HTML.
    CONCATENATE GV_ECO_HTML '<td class="px-4 py-3 text-xs text-slate-700">' lv_eco_name_v '</td>' INTO GV_ECO_HTML.
    CONCATENATE GV_ECO_HTML '<td class="px-4 py-3 text-xs text-slate-500">' ls_eco_v-created_date '</td>' INTO GV_ECO_HTML.
    CONCATENATE GV_ECO_HTML '</tr>' INTO GV_ECO_HTML.
  ENDLOOP.
ELSE.
  CONCATENATE GV_ECO_HTML '<tr><td colspan="8" class="px-4 py-12 text-center">' INTO GV_ECO_HTML.
  CONCATENATE GV_ECO_HTML '<div class="flex flex-col items-center">' INTO GV_ECO_HTML.
  CONCATENATE GV_ECO_HTML '<i class="fas fa-gears text-3xl text-slate-300 mb-3"></i>' INTO GV_ECO_HTML.
  CONCATENATE GV_ECO_HTML '<p class="text-sm text-slate-400 font-medium">Tidak ada ECO</p>' INTO GV_ECO_HTML.
  CONCATENATE GV_ECO_HTML '<p class="text-xs text-slate-300 mt-1">ECO yang sedang diproses akan muncul di sini</p>' INTO GV_ECO_HTML.
  CONCATENATE GV_ECO_HTML '</div></td></tr>' INTO GV_ECO_HTML.
ENDIF.
