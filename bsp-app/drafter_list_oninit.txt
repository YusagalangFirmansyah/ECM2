GV_USER_ID     = request->get_form_field( 'u' ).
GV_FULL_NAME   = request->get_form_field( 'n' ).
GV_ROLE        = request->get_form_field( 'r' ).
LV_FILTER      = request->get_form_field( 'filter' ).
LV_VIEW_MODE   = request->get_form_field( 'view' ).
LV_ECR_DETAIL  = request->get_form_field( 'ecr_detail' ).
GV_MESSAGE     = request->get_form_field( 'msg' ).
GV_ICON        = request->get_form_field( 'icon' ).
GV_DATE_FROM   = request->get_form_field( 'date_from' ).
GV_DATE_TO     = request->get_form_field( 'date_to' ).

" Default date range: first day of current month to today
IF GV_DATE_FROM IS INITIAL.
  " Default: 4 months ago
  DATA: lv_yr TYPE n LENGTH 4, lv_mo TYPE n LENGTH 2, lv_yy TYPE n LENGTH 4, lv_mm TYPE n LENGTH 2.
  lv_yy = sy-datum(4). lv_mm = sy-datum+4(2).
  lv_yr = lv_yy. lv_mo = lv_mm.
  IF lv_mo > 4.
    lv_mo = lv_mo - 4.
  ELSE.
    lv_yr = lv_yr - 1. lv_mo = lv_mo + 8.
  ENDIF.
  CONCATENATE lv_yr lv_mo '01' INTO GV_DATE_FROM.
ENDIF.
IF GV_DATE_TO IS INITIAL.
  GV_DATE_TO = sy-datum.
ENDIF.

" Strip dashes (HTML date input sends YYYY-MM-DD)
REPLACE ALL OCCURRENCES OF '-' IN GV_DATE_FROM WITH ''.
REPLACE ALL OCCURRENCES OF '-' IN GV_DATE_TO WITH ''.

DATA: lv_df_internal TYPE sy-datum,
      lv_dt_internal TYPE sy-datum.
lv_df_internal = GV_DATE_FROM.
lv_dt_internal = GV_DATE_TO.

IF LV_FILTER IS INITIAL.
  LV_FILTER = 'ALL'.
ENDIF.
LV_VIEW_MODE = 'MEINE'.

" Modal state (for material search)
LV_MODAL_OPEN  = request->get_form_field( 'modal' ).
LV_MAT_KEYWORD = request->get_form_field( 'mat_keyword' ).

DATA: lv_proc_act TYPE string.
lv_proc_act = request->get_form_field( 'OnInputProcessing' ).
IF lv_proc_act <> 'search_material'.
  CLEAR: GV_MAT_RESULTS, LV_MAT_KEYWORD, LV_MODAL_OPEN.
ENDIF.

" Status counts (cards) — filtered by drafter
DATA: lt_drafter_status TYPE STANDARD TABLE OF ty_status_count,
      ls_drafter_status TYPE ty_status_count.

IF GV_ROLE = 'DRAFTER_WOOD'.
  SELECT status, COUNT(*) AS cnt
    INTO TABLE @lt_drafter_status
    FROM zecm_req
    WHERE drafter = @GV_USER_ID
      AND created_date >= @lv_df_internal AND created_date <= @lv_dt_internal
      AND ( kategori = 'DRW' OR kategori = 'DBX' )
    GROUP BY status.
ELSEIF GV_ROLE = 'DRAFTER_METAL'.
  SELECT status, COUNT(*) AS cnt
    INTO TABLE @lt_drafter_status
    FROM zecm_req
    WHERE drafter = @GV_USER_ID
      AND created_date >= @lv_df_internal AND created_date <= @lv_dt_internal
      AND kategori = 'DMT'
    GROUP BY status.
ELSE.
  SELECT status, COUNT(*) AS cnt
    INTO TABLE @lt_drafter_status
    FROM zecm_req
    WHERE drafter = @GV_USER_ID
      AND created_date >= @lv_df_internal AND created_date <= @lv_dt_internal
    GROUP BY status.
ENDIF.

LOOP AT lt_drafter_status INTO ls_drafter_status.
  CASE ls_drafter_status-status.
    WHEN 'CRTD'. GV_COUNT_CRTD = ls_drafter_status-count.
    WHEN 'INWK'. GV_COUNT_INWK = ls_drafter_status-count.
    WHEN 'BMPR'. GV_COUNT_BMPR = ls_drafter_status-count.
    WHEN 'BMAP'. GV_COUNT_BMAP = ls_drafter_status-count.
    WHEN 'INSP'. GV_COUNT_INSP = ls_drafter_status-count.
    WHEN 'REJC'. GV_COUNT_REJC = ls_drafter_status-count.
  ENDCASE.
ENDLOOP.

" ECR Detail + Audit Trail

           ON a~ecr_no = i~ecr_no

           ON a~matnr_fg = b~matnr

           ON a~drafter = d~zecm_user_id
         WHERE a~ecr_no = @LV_ECR_DETAIL.

  IF LS_ECR_DETAIL IS NOT INITIAL.
    DATA: lt_det_dir_parts  TYPE STANDARD TABLE OF string,
          lv_det_dokar_dir  TYPE string,

          lv_dokvr_detail   TYPE char2,

    LS_ECR_DETAIL-matnr_fg_no_zero = LS_ECR_DETAIL-matnr_fg.

        ORDER BY att_id ASCENDING.

    IF LS_ECR_DETAIL-dir_number IS NOT INITIAL.
      SPLIT LS_ECR_DETAIL-dir_number AT '/' INTO TABLE lt_det_dir_parts.
      READ TABLE lt_det_dir_parts INTO lv_det_dokar_dir INDEX 1.
      READ TABLE lt_det_dir_parts INTO lv_det_doknr_dir INDEX 2.
      IF lv_det_doknr_dir IS NOT INITIAL.
        SHIFT lv_det_doknr_dir LEFT DELETING LEADING '0'.
        LS_ECR_DETAIL-dir_number_clean = lv_det_doknr_dir.
        LS_ECR_DETAIL-dms_doknr        = lv_det_doknr_dir.
        LS_ECR_DETAIL-dms_dokar        = lv_det_dokar_dir.
        LS_ECR_DETAIL-dms_dokvr        = '00'.
        LS_ECR_DETAIL-dms_doktl        = '000'.
        LS_ECR_DETAIL-dms_found        = abap_true.
      ELSE.
        LS_ECR_DETAIL-dir_number_clean = LS_ECR_DETAIL-dir_number.
      ENDIF.
    ENDIF.

    " Target DMS status for detail
    CLEAR lv_target_dokst_d.
    IF LS_ECR_DETAIL-status = 'INWK' OR LS_ECR_DETAIL-status = 'BMAP'.
      lv_target_dokst_d = 'IA'.
    ELSEIF LS_ECR_DETAIL-status = 'INSP'.
      lv_target_dokst_d = 'AP'.
    ELSEIF LS_ECR_DETAIL-status = 'CLSD'.
      lv_target_dokst_d = 'FR'.
    ENDIF.

    " Prioritas 1: Jika sudah ada dir_number, cek status DRAW dokumen tersebut langsung
    IF LS_ECR_DETAIL-dir_number IS NOT INITIAL AND lv_det_doknr_dir IS NOT INITIAL.
      CLEAR: lv_det_doknr_q, lv_det_dokar_q.
      lv_det_dokar_q = lv_det_dokar_dir.
      lv_det_doknr_q = lv_det_doknr_dir.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
        EXPORTING  input  = lv_det_doknr_q
        IMPORTING  output = lv_det_doknr_q.

      SELECT SINGLE dokvr, doktl, dokst
        FROM draw
        INTO (@LS_ECR_DETAIL-dms_dokvr, @LS_ECR_DETAIL-dms_doktl, @LS_ECR_DETAIL-dms_dokst)
        WHERE dokar = @lv_det_dokar_q
          AND doknr = @lv_det_doknr_q.
      IF sy-subrc = 0.
        SELECT SINGLE dktxt FROM drat INTO @LS_ECR_DETAIL-dms_dktxt
          WHERE dokar = @lv_det_dokar_q
            AND doknr = @lv_det_doknr_q
            AND dokvr = @LS_ECR_DETAIL-dms_dokvr
            AND doktl = @LS_ECR_DETAIL-dms_doktl.
      ENDIF.
    ENDIF.

    " Prioritas 2: DMS lookup — search DRAD by material jika belum ada dokumen
    IF LS_ECR_DETAIL-dms_doknr IS INITIAL AND
       ( ( ( LS_ECR_DETAIL-status = 'INWK' OR LS_ECR_DETAIL-status = 'BMAP' ) AND LS_ECR_DETAIL-bom_check = 'A' )
         OR ( ( LS_ECR_DETAIL-status = 'INSP' OR LS_ECR_DETAIL-status = 'CLSD' )
              AND ( LS_ECR_DETAIL-bom_check = 'A' OR LS_ECR_DETAIL-bom_check IS INITIAL ) ) ).
      lv_matnr_detail = LS_ECR_DETAIL-matnr_fg.
      CLEAR: lv_doknr_detail, lv_dokar_detail, lv_dokvr_detail, lv_doktl_detail, lv_dokst_detail.

      SELECT SINGLE a~doknr, a~dokar, a~dokvr, a~doktl, b~dokst
        INTO (@lv_doknr_detail, @lv_dokar_detail, @lv_dokvr_detail, @lv_doktl_detail, @lv_dokst_detail)
        FROM drad AS a
        INNER JOIN draw AS b
          ON a~dokar = b~dokar
         AND a~doknr = b~doknr
         AND a~dokvr = b~dokvr
         AND a~doktl = b~doktl
        WHERE a~objky = @lv_matnr_detail
          AND a~dokob = 'MARA'
          AND a~dokar = @LS_ECR_DETAIL-kategori
          AND b~dokst = @lv_target_dokst_d.

      " Fallback: coba dengan ALPHA_INPUT (kalau DRAD simpan dengan leading zeros)
      IF sy-subrc <> 0.
        CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
          EXPORTING  input  = lv_matnr_detail
          IMPORTING  output = lv_matnr_detail.

        SELECT SINGLE a~doknr, a~dokar, a~dokvr, a~doktl, b~dokst
          INTO (@lv_doknr_detail, @lv_dokar_detail, @lv_dokvr_detail, @lv_doktl_detail, @lv_dokst_detail)
          FROM drad AS a
          INNER JOIN draw AS b
            ON a~dokar = b~dokar
           AND a~doknr = b~doknr
           AND a~dokvr = b~dokvr
           AND a~doktl = b~doktl
          WHERE a~objky = @lv_matnr_detail
            AND a~dokob = 'MARA'
            AND a~dokar = @LS_ECR_DETAIL-kategori
            AND b~dokst = @lv_target_dokst_d.
      ENDIF.

      IF sy-subrc = 0.
        LS_ECR_DETAIL-dms_doknr = lv_doknr_detail.
        LS_ECR_DETAIL-dms_dokar = lv_dokar_detail.
        LS_ECR_DETAIL-dms_dokvr = lv_dokvr_detail.
        LS_ECR_DETAIL-dms_doktl = lv_doktl_detail.
        LS_ECR_DETAIL-dms_dokst = lv_dokst_detail.
      ENDIF.

      IF LS_ECR_DETAIL-dms_doknr IS NOT INITIAL.
        LS_ECR_DETAIL-dms_found = abap_true.
        SELECT SINGLE dktxt FROM drat INTO @LS_ECR_DETAIL-dms_dktxt
          WHERE dokar = @LS_ECR_DETAIL-dms_dokar
            AND doknr = @LS_ECR_DETAIL-dms_doknr
            AND dokvr = @LS_ECR_DETAIL-dms_dokvr
            AND doktl = @LS_ECR_DETAIL-dms_doktl.
        CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
          EXPORTING  input  = LS_ECR_DETAIL-dms_doknr
          IMPORTING  output = LS_ECR_DETAIL-dms_doknr.
      ENDIF.
    ENDIF.
  ENDIF.

  " Audit Trail
  SELECT log_id, ecr_no, action, old_status, new_status,
         user_id, e~zecm_full_name AS user_name, action_date, action_time, reason
         INTO TABLE @LT_LOGS
         FROM zecm_log
         LEFT OUTER JOIN zecm_user AS e
           ON zecm_log~user_id = e~zecm_user_id
         WHERE ecr_no = @LV_ECR_DETAIL
        AND is_ireq = ''
         ORDER BY action_date DESCENDING, action_time DESCENDING.

  LOOP AT LT_LOGS INTO LS_LOG.
    CASE LS_LOG-old_status.
      WHEN 'CRTD'. LS_LOG-old_label = 'Draft'.
      WHEN 'INWK'. LS_LOG-old_label = 'In Work'.
      WHEN 'INSP'. LS_LOG-old_label = 'Inspection'.
      WHEN 'CLSD'. LS_LOG-old_label = 'Closed'.
      WHEN 'REJC'. LS_LOG-old_label = 'Rejected'.
      WHEN OTHERS. LS_LOG-old_label = LS_LOG-old_status.
    ENDCASE.
    CASE LS_LOG-new_status.
      WHEN 'CRTD'. LS_LOG-new_label = 'Draft'.
      WHEN 'INWK'. LS_LOG-new_label = 'In Work'.
      WHEN 'INSP'. LS_LOG-new_label = 'Inspection'.
      WHEN 'CLSD'. LS_LOG-new_label = 'Closed'.
      WHEN 'REJC'. LS_LOG-new_label = 'Rejected'.
      WHEN OTHERS. LS_LOG-new_label = LS_LOG-new_status.
    ENDCASE.
    CASE LS_LOG-action.
      WHEN 'CREA'. LS_LOG-action_label = 'Created'.
      WHEN 'APPR'. LS_LOG-action_label = 'Approved'.
      WHEN 'INSP'. LS_LOG-action_label = 'Submitted'.
      WHEN 'REJT'. LS_LOG-action_label = 'Rejected'.
      WHEN 'CANC'. LS_LOG-action_label = 'Cancelled'.
      WHEN 'BOM_PENDING'. LS_LOG-action_label = 'BOM Uploaded'.
      WHEN 'BOM_APPROVED'. LS_LOG-action_label = 'BOM Approved'.
      WHEN 'BOM_REJECTED'. LS_LOG-action_label = 'BOM Rejected'.
      WHEN OTHERS. LS_LOG-action_label = LS_LOG-action.
    ENDCASE.
    MODIFY LT_LOGS FROM LS_LOG.
  ENDLOOP.

ELSE.

  " List view — filter by view mode + status
  IF LV_VIEW_MODE = 'MEINE'.
    " ECR Saya (default)
    IF LV_FILTER = 'ALL'.
      SELECT a~ecr_no, a~title, a~matnr_fg, a~kategori, a~priority,
             a~status, a~requester, a~drafter, a~descr, a~dokar, a~dir_number,
             a~reject_reason, a~cost_estimate, a~zdept, a~zreason,
             a~created_date, a~changed_date,
             a~bom_check, a~bom_notes,
             c~zecm_full_name AS requester_name, d~zecm_full_name AS drafter_name, b~maktx,
             i~impl_po, i~impl_date, i~impl_reason
             INTO CORRESPONDING FIELDS OF TABLE @LT_ECRS
             FROM zecm_req AS a
             LEFT OUTER JOIN zecm_ireq AS i
               ON a~ecr_no = i~ecr_no
             LEFT OUTER JOIN makt AS b
               ON a~matnr_fg = b~matnr
              AND b~spras = @sy-langu
             LEFT OUTER JOIN zecm_user AS c
               ON a~requester = c~zecm_user_id
             LEFT OUTER JOIN zecm_user AS d
               ON a~drafter = d~zecm_user_id
             WHERE a~drafter = @GV_USER_ID
               AND a~created_date >= @lv_df_internal AND a~created_date <= @lv_dt_internal
             ORDER BY a~created_date DESCENDING.
    ELSE.
      SELECT a~ecr_no, a~title, a~matnr_fg, a~kategori, a~priority,
             a~status, a~requester, a~drafter, a~descr, a~dokar, a~dir_number,
             a~reject_reason, a~cost_estimate, a~zdept, a~zreason,
             a~created_date, a~changed_date,
             a~bom_check, a~bom_notes,
             c~zecm_full_name AS requester_name, d~zecm_full_name AS drafter_name, b~maktx,
             i~impl_po, i~impl_date, i~impl_reason
             INTO CORRESPONDING FIELDS OF TABLE @LT_ECRS
             FROM zecm_req AS a
             LEFT OUTER JOIN zecm_ireq AS i
               ON a~ecr_no = i~ecr_no
             LEFT OUTER JOIN makt AS b
               ON a~matnr_fg = b~matnr
              AND b~spras = @sy-langu
             LEFT OUTER JOIN zecm_user AS c
               ON a~requester = c~zecm_user_id
             LEFT OUTER JOIN zecm_user AS d
               ON a~drafter = d~zecm_user_id
             WHERE a~drafter = @GV_USER_ID
               AND a~status = @LV_FILTER
               AND a~created_date >= @lv_df_internal AND a~created_date <= @lv_dt_internal
             ORDER BY a~created_date DESCENDING.
    ENDIF.
  ELSE.
    " Tampilkan Semua
    IF LV_FILTER = 'ALL'.
      SELECT a~ecr_no, a~title, a~matnr_fg, a~kategori, a~priority,
             a~status, a~requester, a~drafter, a~descr, a~dokar, a~dir_number,
             a~reject_reason, a~cost_estimate, a~zdept, a~zreason,
             a~created_date, a~changed_date,
             a~bom_check, a~bom_notes,
             c~zecm_full_name AS requester_name, d~zecm_full_name AS drafter_name, b~maktx,
             i~impl_po, i~impl_date, i~impl_reason
             INTO CORRESPONDING FIELDS OF TABLE @LT_ECRS
             FROM zecm_req AS a
             LEFT OUTER JOIN zecm_ireq AS i
               ON a~ecr_no = i~ecr_no
             LEFT OUTER JOIN makt AS b
               ON a~matnr_fg = b~matnr
              AND b~spras = @sy-langu
             LEFT OUTER JOIN zecm_user AS c
               ON a~requester = c~zecm_user_id
             LEFT OUTER JOIN zecm_user AS d
               ON a~drafter = d~zecm_user_id
             WHERE a~created_date >= @lv_df_internal AND a~created_date <= @lv_dt_internal
             ORDER BY a~created_date DESCENDING.
    ELSE.
      SELECT a~ecr_no, a~title, a~matnr_fg, a~kategori, a~priority,
             a~status, a~requester, a~drafter, a~descr, a~dokar, a~dir_number,
             a~reject_reason, a~cost_estimate, a~zdept, a~zreason,
             a~created_date, a~changed_date,
             a~bom_check, a~bom_notes,
             c~zecm_full_name AS requester_name, d~zecm_full_name AS drafter_name, b~maktx,
             i~impl_po, i~impl_date, i~impl_reason
             INTO CORRESPONDING FIELDS OF TABLE @LT_ECRS
             FROM zecm_req AS a
             LEFT OUTER JOIN zecm_ireq AS i
               ON a~ecr_no = i~ecr_no
             LEFT OUTER JOIN makt AS b
               ON a~matnr_fg = b~matnr
              AND b~spras = @sy-langu
             LEFT OUTER JOIN zecm_user AS c
               ON a~requester = c~zecm_user_id
             LEFT OUTER JOIN zecm_user AS d
               ON a~drafter = d~zecm_user_id
             WHERE a~status = @LV_FILTER
               AND a~created_date >= @lv_df_internal AND a~created_date <= @lv_dt_internal
             ORDER BY a~created_date DESCENDING.
    ENDIF.
  ENDIF.

  " Filter division for Wood and Metal Drafters
  IF GV_ROLE = 'DRAFTER_WOOD'.
    DELETE LT_ECRS WHERE kategori <> 'DRW' AND kategori <> 'DBX'.
  ELSEIF GV_ROLE = 'DRAFTER_METAL'.
    DELETE LT_ECRS WHERE kategori <> 'DMT'.
  ENDIF.

  " Remove leading zeros + DMS detection — search DRAD by material
  DATA: lv_dir_parts    TYPE STANDARD TABLE OF string,
        lv_dir_dokar    TYPE string,
        lv_dir_doknr    TYPE string,
        lv_q_dokar      TYPE dokar,
        lv_q_doknr      TYPE doknr,
        lv_target_dokst TYPE char2,
        lv_matnr_conv   TYPE matnr,
        lv_doknr_find   TYPE doknr,
        lv_dokar_find   TYPE dokar,
        lv_dokvr_find   TYPE char2,
        lv_doktl_find   TYPE char3,
        lv_dokst_find   TYPE char2.

  LOOP AT LT_ECRS INTO LS_ECR.
    LS_ECR-matnr_fg_no_zero = LS_ECR-matnr_fg.
    SHIFT LS_ECR-matnr_fg_no_zero LEFT DELETING LEADING '0'.

    " Clean dir_number (remove leading zeros from doknr part)
    CLEAR: lv_dir_dokar, lv_dir_doknr, lv_q_dokar, lv_q_doknr.
    IF LS_ECR-dir_number IS NOT INITIAL.
      CLEAR lv_dir_parts.
      SPLIT LS_ECR-dir_number AT '/' INTO TABLE lv_dir_parts.
      READ TABLE lv_dir_parts INTO lv_dir_dokar INDEX 1.
      READ TABLE lv_dir_parts INTO lv_dir_doknr INDEX 2.
      IF lv_dir_doknr IS NOT INITIAL.
        SHIFT lv_dir_doknr LEFT DELETING LEADING '0'.
        LS_ECR-dir_number_clean = lv_dir_doknr.
      ELSE.
        LS_ECR-dir_number_clean = LS_ECR-dir_number.
      ENDIF.
    ENDIF.

    " Prioritas 1: Jika ECR sudah memiliki dir_number, cek status dokumen di DRAW
    IF LS_ECR-dir_number IS NOT INITIAL AND lv_dir_doknr IS NOT INITIAL.
      lv_q_dokar = lv_dir_dokar.
      lv_q_doknr = lv_dir_doknr.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
        EXPORTING  input  = lv_q_doknr
        IMPORTING  output = lv_q_doknr.

      SELECT SINGLE dokvr, doktl, dokst
        FROM draw
        INTO (@LS_ECR-dms_dokvr, @LS_ECR-dms_doktl, @LS_ECR-dms_dokst)
        WHERE dokar = @lv_q_dokar
          AND doknr = @lv_q_doknr.

      IF sy-subrc = 0.
        LS_ECR-dms_doknr = lv_dir_doknr.
        LS_ECR-dms_dokar = lv_q_dokar.
        LS_ECR-dms_found = abap_true.
        SELECT SINGLE dktxt FROM drat INTO @LS_ECR-dms_dktxt
          WHERE dokar = @lv_q_dokar
            AND doknr = @lv_q_doknr
            AND dokvr = @LS_ECR-dms_dokvr
            AND doktl = @LS_ECR-dms_doktl.
      ENDIF.
    ENDIF.

    " Target DMS status
    CLEAR lv_target_dokst.
    IF LS_ECR-status = 'INWK' OR LS_ECR-status = 'BMAP'.
      lv_target_dokst = 'IA'.
    ELSEIF LS_ECR-status = 'INSP'.
      lv_target_dokst = 'AP'.
    ELSEIF LS_ECR-status = 'CLSD'.
      lv_target_dokst = 'FR'.
    ENDIF.

    " Prioritas 2: DMS lookup via DRAD jika belum ada dir_number terdaftar
    IF LS_ECR-dms_doknr IS INITIAL.
      " BOM Gate: INWK only detect if bom_check = 'A', INSP/CLSD backward compat
      IF ( ( LS_ECR-status = 'INWK' OR LS_ECR-status = 'BMAP' ) AND LS_ECR-bom_check = 'A' )
         OR ( ( LS_ECR-status = 'INSP' OR LS_ECR-status = 'CLSD' )
              AND ( LS_ECR-bom_check = 'A' OR LS_ECR-bom_check IS INITIAL ) ).
        lv_matnr_conv = LS_ECR-matnr_fg.

        CLEAR: lv_doknr_find, lv_dokar_find, lv_dokvr_find, lv_doktl_find, lv_dokst_find.

        " Try without ALPHA first (DRAD might store without leading zeros)
        SELECT SINGLE a~doknr, a~dokar, a~dokvr, a~doktl, b~dokst
          INTO (@lv_doknr_find, @lv_dokar_find, @lv_dokvr_find, @lv_doktl_find, @lv_dokst_find)
          FROM drad AS a
          INNER JOIN draw AS b
            ON a~dokar = b~dokar
           AND a~doknr = b~doknr
           AND a~dokvr = b~dokvr
           AND a~doktl = b~doktl
          WHERE a~objky = @lv_matnr_conv
            AND a~dokob = 'MARA'
            AND a~dokar = @LS_ECR-kategori
            AND b~dokst = @lv_target_dokst.

        " Fallback: coba dengan ALPHA_INPUT (kalau DRAD simpan dengan leading zeros)
        IF sy-subrc <> 0.
          CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
            EXPORTING  input  = lv_matnr_conv
            IMPORTING  output = lv_matnr_conv.

          SELECT SINGLE a~doknr, a~dokar, a~dokvr, a~doktl, b~dokst
            INTO (@lv_doknr_find, @lv_dokar_find, @lv_dokvr_find, @lv_doktl_find, @lv_dokst_find)
            FROM drad AS a
            INNER JOIN draw AS b
              ON a~dokar = b~dokar
             AND a~doknr = b~doknr
             AND a~dokvr = b~dokvr
             AND a~doktl = b~doktl
            WHERE a~objky = @lv_matnr_conv
              AND a~dokob = 'MARA'
              AND a~dokar = @LS_ECR-kategori
              AND b~dokst = @lv_target_dokst.
        ENDIF.

        IF sy-subrc = 0.
          LS_ECR-dms_doknr = lv_doknr_find.
          LS_ECR-dms_dokar = lv_dokar_find.
          LS_ECR-dms_dokvr = lv_dokvr_find.
          LS_ECR-dms_doktl = lv_doktl_find.
          LS_ECR-dms_dokst = lv_dokst_find.
        ENDIF.

        IF LS_ECR-dms_doknr IS NOT INITIAL.
          LS_ECR-dms_found = abap_true.
          SELECT SINGLE dktxt FROM drat INTO @LS_ECR-dms_dktxt
            WHERE dokar = @LS_ECR-dms_dokar
              AND doknr = @LS_ECR-dms_doknr
              AND dokvr = @LS_ECR-dms_dokvr
              AND doktl = @LS_ECR-dms_doktl.
          CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
            EXPORTING  input  = LS_ECR-dms_doknr
            IMPORTING  output = LS_ECR-dms_doknr.
        ENDIF.
      ENDIF.
    ENDIF.

    MODIFY LT_ECRS FROM LS_ECR.
  ENDLOOP.

ENDIF.
