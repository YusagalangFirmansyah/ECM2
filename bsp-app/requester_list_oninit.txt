" --- 2. Material Search (OnInit — langsung dari form) ---
GV_MAT_KEYWORD = request->get_form_field( 'keyword' ).
IF GV_MAT_KEYWORD IS NOT INITIAL.
  CLEAR GV_MAT_RESULTS.
  CONCATENATE '%' GV_MAT_KEYWORD '%' INTO GV_MAT_KEYWORD.
  TRANSLATE GV_MAT_KEYWORD TO UPPER CASE.
  SELECT a~matnr, b~maktx, a~mtart UP TO 20 ROWS
    INTO CORRESPONDING FIELDS OF TABLE @LT_MATERIAL
    FROM MARA AS A
    INNER JOIN MAKT AS B
      ON B~matnr = A~matnr AND B~spras = @SY-LANGU
    WHERE A~mtart IN ('FERT', 'HALB', 'HALM')
      AND ( A~matnr LIKE @GV_MAT_KEYWORD
         OR B~maktx LIKE @GV_MAT_KEYWORD ).
  IF SY-SUBRC = 0.
    SORT LT_MATERIAL BY MTART ASCENDING.
    DATA: lv_maktx_safe TYPE string.
    LOOP AT LT_MATERIAL INTO LS_MATERIAL.
      CASE LS_MATERIAL-MTART.
        WHEN 'FERT'.
          LV_TLABEL = 'Finish Goods'.
          LV_TCOLOR = 'bg-emerald-100 text-emerald-700'.
        WHEN 'HALB'.
          LV_TLABEL = 'Semifinish Goods'.
          LV_TCOLOR = 'bg-blue-100 text-blue-700'.
        WHEN 'HALM'.
          LV_TLABEL = 'Semifinish Metal'.
          LV_TCOLOR = 'bg-purple-100 text-purple-700'.
        WHEN OTHERS.
          LV_TLABEL = LS_MATERIAL-MTART.
          LV_TCOLOR = 'bg-slate-100 text-slate-600'.
      ENDCASE.

      " Encode maktx for safe use in data-* attributes
      lv_maktx_safe = LS_MATERIAL-MAKTX.
      REPLACE ALL OCCURRENCES OF '&' IN lv_maktx_safe WITH '&amp;'.
      REPLACE ALL OCCURRENCES OF '"' IN lv_maktx_safe WITH '&quot;'.

      CONCATENATE GV_MAT_RESULTS
        '<div class="mat-pick" style="display:flex;align-items:center;justify-content:space-between;cursor:pointer;'
        'padding:6px 12px;border-bottom:1px solid #e2e8f0;"'
        ' data-matnr="' LS_MATERIAL-MATNR '"'
        ' data-maktx="' lv_maktx_safe '">'
        '<div style="display:flex;align-items:center;gap:8px;">'
        '<span class="inline-block px-2 py-1 rounded-md text-[0.65rem] font-bold tracking-wide '
        'uppercase border ' LV_TCOLOR '">' LV_TLABEL '</span>'
        '<span class="text-xs font-semibold text-slate-800">' LS_MATERIAL-MATNR '</span>'
        '<span class="text-xs text-slate-400">' LS_MATERIAL-MAKTX '</span>'
        '</div>'
        '<button type="button"'
        ' class="px-2 py-1 bg-emerald-100 text-emerald-700 rounded text-xs font-bold hover:bg-emerald-200 transition">'

        INTO GV_MAT_RESULTS.
    ENDLOOP.
  ELSE.
    GV_MAT_RESULTS = '<div class="p-3 text-sm text-slate-400 text-center">'
      && '<i class="fas fa-search text-2xl mb-2 block"></i>'
      && 'Material tidak ditemukan.</div>'.
  ENDIF.
ENDIF.

" --- 3. Submit Request (inline di OnInit) ---
IF LV_ACTION IS INITIAL.
  LV_ACTION = request->get_form_field( 'OnInputProcessing' ).
ENDIF.
IF LV_ACTION = 'submit_request'.
  DATA: LV_MAT_LIST TYPE string,
        LV_TITLE_VAL TYPE string,
        LV_DESC_VAL TYPE string,
        LV_KAT_VAL TYPE string,
        LV_PRI_VAL TYPE string,
        LV_DEPT_VAL TYPE string,
        LV_REASON_VAL TYPE string,
        LV_IREQ_NUM TYPE numc10,
        LS_NEW_IREQ TYPE zecm_ireq,
        LS_NEW_LOG TYPE zecm_log.

  LV_MAT_LIST   = request->get_form_field( 'materials' ).
  LV_TITLE_VAL  = request->get_form_field( 'title' ).
  LV_DESC_VAL   = request->get_form_field( 'description' ).
  LV_KAT_VAL    = request->get_form_field( 'kategori' ).
  LV_PRI_VAL    = request->get_form_field( 'priority' ).
  LV_DEPT_VAL   = request->get_form_field( 'dept' ).
  LV_REASON_VAL = request->get_form_field( 'reason' ).

  IF GV_USER_ID IS INITIAL.
    GV_USER_ID = request->get_form_field( 'u' ).
  ENDIF.

  IF LV_TITLE_VAL IS INITIAL.
    GV_MESSAGE = 'Judul wajib diisi.'. GV_ICON = 'error'.
  ELSEIF LV_MAT_LIST IS INITIAL.
    GV_MESSAGE = 'Pilih minimal satu material.'. GV_ICON = 'error'.
  ELSEIF LV_KAT_VAL IS INITIAL.
    GV_MESSAGE = 'Kategori wajib dipilih.'. GV_ICON = 'error'.
  ELSEIF LV_PRI_VAL IS INITIAL.
    GV_MESSAGE = 'Prioritas wajib dipilih.'. GV_ICON = 'error'.
  ELSEIF LV_DEPT_VAL IS INITIAL.
    GV_MESSAGE = 'Departemen wajib dipilih.'. GV_ICON = 'error'.
  ELSEIF LV_REASON_VAL IS INITIAL.
    GV_MESSAGE = 'Alasan wajib dipilih.'. GV_ICON = 'error'.
  ELSE.
    SPLIT LV_MAT_LIST AT ',' INTO TABLE DATA(LT_SEL).
    READ TABLE LT_SEL INTO LV_MAT_LIST INDEX 1.
    CONDENSE LV_MAT_LIST NO-GAPS.

    CALL FUNCTION 'NUMBER_GET_NEXT'
      EXPORTING nr_range_nr = '01' object = 'ZECM_IREQ'
      IMPORTING number = LV_IREQ_NUM
      EXCEPTIONS OTHERS = 1.

    IF sy-subrc <> 0.
      GV_MESSAGE = 'Gagal generate nomor request.'. GV_ICON = 'error'.
    ELSE.
      CLEAR LS_NEW_IREQ.
      LS_NEW_IREQ-ireq_id    = LV_IREQ_NUM.
      LS_NEW_IREQ-title      = LV_TITLE_VAL.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
        EXPORTING  input  = LV_MAT_LIST
        IMPORTING  output = LV_MAT_LIST.
      CONDENSE LV_MAT_LIST NO-GAPS.
      LS_NEW_IREQ-matnr_fg   = LV_MAT_LIST.
      LS_NEW_IREQ-kategori   = LV_KAT_VAL.
      LS_NEW_IREQ-priority   = LV_PRI_VAL.
      LS_NEW_IREQ-zdept      = LV_DEPT_VAL.
      LS_NEW_IREQ-zreason    = LV_REASON_VAL.
      LS_NEW_IREQ-description = LV_DESC_VAL.
      LS_NEW_IREQ-requester  = GV_USER_ID.
      LS_NEW_IREQ-status     = 'REQ'.
      LS_NEW_IREQ-req_date   = sy-datum.

      INSERT zecm_ireq FROM LS_NEW_IREQ.

      IF sy-subrc = 0.
        CALL FUNCTION 'NUMBER_GET_NEXT'
          EXPORTING nr_range_nr = '01' object = 'ZECM_LOG'
          IMPORTING number = LS_NEW_LOG-log_id
          EXCEPTIONS OTHERS = 1.
        IF sy-subrc = 0.
          LS_NEW_LOG-mandt       = sy-mandt.
          LS_NEW_LOG-ecr_no      = LV_IREQ_NUM.
          LS_NEW_LOG-action      = 'CREA'.
          LS_NEW_LOG-old_status  = ''.
          LS_NEW_LOG-new_status  = 'REQ'.
          LS_NEW_LOG-user_id     = GV_USER_ID.
          LS_NEW_LOG-action_date = sy-datum.
          LS_NEW_LOG-action_time = sy-uzeit.
          LS_NEW_LOG-is_ireq    = 'X'.
          INSERT zecm_log FROM LS_NEW_LOG.
        ENDIF.

        LV_MSG = 'Request berhasil dikirim.'.
        REPLACE ALL OCCURRENCES OF REGEX '[ ]' IN LV_MSG WITH '%20'.
        CONCATENATE 'requester_list.htm?u=' GV_USER_ID '&msg=' LV_MSG INTO DATA(LV_URL).
        navigation->goto_page( LV_URL ).
        RETURN.
      ELSE.
        GV_MESSAGE = 'Gagal menyimpan request.'. GV_ICON = 'error'.
      ENDIF.
    ENDIF.
  ENDIF.
ENDIF.

" --- 4. Load Request List (with filters) ---

  SELECT * FROM zecm_ireq
    INTO CORRESPONDING FIELDS OF TABLE @LT_REQUEST
    WHERE requester = @GV_USER_ID
    ORDER BY req_date DESCENDING, ireq_id DESCENDING.
ELSEIF GV_SEARCH IS NOT INITIAL.
  DATA: lv_search_kw TYPE string.
  lv_search_kw = GV_SEARCH.
  TRANSLATE lv_search_kw TO UPPER CASE.
  CONCATENATE '%' lv_search_kw '%' INTO lv_search_kw.
  SELECT * FROM zecm_ireq
    INTO CORRESPONDING FIELDS OF TABLE @LT_REQUEST
    WHERE requester = @GV_USER_ID
      AND ( title LIKE @lv_search_kw OR matnr_fg LIKE @lv_search_kw )
    ORDER BY req_date DESCENDING, ireq_id DESCENDING.
ELSE.
  SELECT * FROM zecm_ireq
    INTO CORRESPONDING FIELDS OF TABLE @LT_REQUEST
    WHERE requester = @GV_USER_ID
    ORDER BY req_date DESCENDING, ireq_id DESCENDING.
ENDIF.
