  INTO CORRESPONDING FIELDS OF TABLE @lt_drafter_c
  FROM zecm_user
  WHERE zecm_role = 'DRAFTER' AND zecm_active = 'Y'
  ORDER BY zecm_full_name ASCENDING.

" 2. Ambil ECR Aktif yang sedang ditangani (Bukan CRTD, CLSD, REJC)
SELECT drafter, status
  INTO CORRESPONDING FIELDS OF TABLE @lt_ecr_c
  FROM zecm_req
  WHERE ( status = 'INWK' OR status = 'BMPR' OR status = 'BMAP' OR status = 'INSP' )
    AND drafter <> ''.

DATA: lv_c_inwk TYPE i, lv_c_bmpr TYPE i, lv_c_bmap TYPE i, lv_c_insp TYPE i.
DATA: lv_str_inwk TYPE string, lv_str_bmpr TYPE string, lv_str_bmap TYPE string, lv_str_insp TYPE string.
DATA: lv_c_labels TYPE string, lv_c_dinwk TYPE string, lv_c_dbmpr TYPE string, lv_c_dbmap TYPE string, lv_c_dinsp TYPE string.
DATA: lv_c_comma TYPE string VALUE ''.

  CLEAR: lv_c_inwk, lv_c_bmpr, lv_c_bmap, lv_c_insp.

  LOOP AT lt_ecr_c INTO ls_ecr_c WHERE drafter = ls_drafter_c-zecm_user_id.
    CASE ls_ecr_c-status.
      WHEN 'INWK'. lv_c_inwk = lv_c_inwk + 1.
      WHEN 'BMPR'. lv_c_bmpr = lv_c_bmpr + 1.
      WHEN 'BMAP'. lv_c_bmap = lv_c_bmap + 1.
      WHEN 'INSP'. lv_c_insp = lv_c_insp + 1.
    ENDCASE.
  ENDLOOP.

  lv_str_inwk = lv_c_inwk. SHIFT lv_str_inwk LEFT DELETING LEADING ' '.
  lv_str_bmpr = lv_c_bmpr. SHIFT lv_str_bmpr LEFT DELETING LEADING ' '.
  lv_str_bmap = lv_c_bmap. SHIFT lv_str_bmap LEFT DELETING LEADING ' '.
  lv_str_insp = lv_c_insp. SHIFT lv_str_insp LEFT DELETING LEADING ' '.

  " Escape name
  DATA: lv_esc_name TYPE string.
  lv_esc_name = ls_drafter_c-zecm_full_name.
  REPLACE ALL OCCURRENCES OF '''' IN lv_esc_name WITH ''.

  CONCATENATE lv_c_labels lv_c_comma '''' lv_esc_name '''' INTO lv_c_labels.
  CONCATENATE lv_c_dinwk lv_c_comma lv_str_inwk INTO lv_c_dinwk.
  CONCATENATE lv_c_dbmpr lv_c_comma lv_str_bmpr INTO lv_c_dbmpr.
  CONCATENATE lv_c_dbmap lv_c_comma lv_str_bmap INTO lv_c_dbmap.
  CONCATENATE lv_c_dinsp lv_c_comma lv_str_insp INTO lv_c_dinsp.
  lv_c_comma = ','.
ENDLOOP.

" ================================================================

GV_MAT_SEARCH = request->get_form_field( 'mat_search' ).

IF GV_MAT_SEARCH IS NOT INITIAL.
  DATA: lt_mat_hist TYPE tt_ecr,
        ls_mat_hist TYPE ty_ecr,
        lv_kw       TYPE string,
        lv_kw_upper TYPE string,
        lv_mat_html TYPE string.

  lv_kw = GV_MAT_SEARCH.
  TRANSLATE lv_kw TO UPPER CASE.
  CONCATENATE '%' lv_kw '%' INTO lv_kw.

  SELECT a~ecr_no, a~title, a~matnr_fg, a~descr, a~kategori, a~priority,
         a~requester, a~drafter, a~dokar, a~dir_number, a~status,
         a~reject_reason, a~cost_estimate, a~zdept, a~zreason,
         a~created_date, a~created_time,
         c~zecm_full_name AS requester_name,
         d~zecm_full_name AS drafter_name,
         b~maktx
    INTO CORRESPONDING FIELDS OF TABLE @lt_mat_hist
    FROM zecm_req AS a
    LEFT OUTER JOIN makt AS b
      ON a~matnr_fg = b~matnr AND b~spras = @sy-langu
    LEFT OUTER JOIN zecm_user AS c
      ON a~requester = c~zecm_user_id
    LEFT OUTER JOIN zecm_user AS d
      ON a~drafter = d~zecm_user_id
    WHERE ( a~matnr_fg LIKE @lv_kw OR b~maktx LIKE @lv_kw )
    ORDER BY a~created_date DESCENDING, a~created_time DESCENDING.

  " Build HTML cards
  DATA: lv_s_matnr   TYPE string,
        lv_s_ecr     TYPE string,
        lv_s_status  TYPE string,
        lv_s_badge   TYPE string,
        lv_s_pri     TYPE string,
        lv_s_pri_clr TYPE string,
        lv_s_kat     TYPE string,
        lv_s_kat_clr TYPE string,
        lv_s_date    TYPE string,
        lv_s_cost    TYPE string,
        lv_s_dms     TYPE string,
        lv_s_reject  TYPE string,
        lv_s_dept    TYPE string,
        lv_s_reason  TYPE string,
        lv_s_drafter TYPE string,
        lv_s_req     TYPE string.

  CLEAR lv_mat_html.


  LOOP AT lt_mat_hist INTO ls_mat_hist.
    " Clean material number
    ls_mat_hist-matnr_fg_no_zero = ls_mat_hist-matnr_fg.
    SHIFT ls_mat_hist-matnr_fg_no_zero LEFT DELETING LEADING '0'.
    
    " Fetch description with ALPHA_INPUT if JOIN failed
    IF ls_mat_hist-maktx IS INITIAL.
      DATA: lv_fix_matnr_hist TYPE matnr.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
        EXPORTING input = ls_mat_hist-matnr_fg
        IMPORTING output = lv_fix_matnr_hist.
      SELECT SINGLE maktx FROM makt INTO @ls_mat_hist-maktx
        WHERE matnr = @lv_fix_matnr_hist AND spras = @sy-langu.
    ENDIF.

    " ECR number clean

    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
      EXPORTING  input  = ls_mat_hist-ecr_no
      IMPORTING  output = ls_mat_hist-ecr_no.

    " DMS detection (any status, simplified)
    IF ls_mat_hist-dir_number IS NOT INITIAL.
      DATA: lv_dir_parts_m TYPE STANDARD TABLE OF string,
            lv_dir_doknr_m TYPE string.
      CLEAR lv_dir_parts_m.
      SPLIT ls_mat_hist-dir_number AT '/' INTO TABLE lv_dir_parts_m.
      READ TABLE lv_dir_parts_m INTO lv_dir_doknr_m INDEX 2.
      IF lv_dir_doknr_m IS NOT INITIAL.
        SHIFT lv_dir_doknr_m LEFT DELETING LEADING '0'.
        ls_mat_hist-dir_number_clean = lv_dir_doknr_m.
      ELSE.
        ls_mat_hist-dir_number_clean = ls_mat_hist-dir_number.
      ENDIF.
    ENDIF.

    " Status badge color
    CASE ls_mat_hist-status.
      WHEN 'CRTD'. lv_s_status = 'Draft'.       lv_s_badge = 'bg-blue-100 text-blue-700'.
      WHEN 'INWK'. lv_s_status = 'In Work'.     lv_s_badge = 'bg-orange-100 text-orange-700'.
      WHEN 'INSP'. lv_s_status = 'Inspection'.  lv_s_badge = 'bg-amber-100 text-amber-700'.
      WHEN 'CLSD'. lv_s_status = 'Closed'.      lv_s_badge = 'bg-green-100 text-green-700'.
      WHEN 'REJC'. lv_s_status = 'Rejected'.    lv_s_badge = 'bg-red-100 text-red-700'.
      WHEN OTHERS. lv_s_status = ls_mat_hist-status. lv_s_badge = 'bg-slate-100 text-slate-600'.
    ENDCASE.

    " Priority color
    CASE ls_mat_hist-priority.
      WHEN 'CRITICAL'. lv_s_pri_clr = 'bg-red-100 text-red-700'.
      WHEN 'HIGH'.     lv_s_pri_clr = 'bg-orange-100 text-orange-700'.
      WHEN 'MEDIUM'.   lv_s_pri_clr = 'bg-blue-100 text-blue-700'.
      WHEN OTHERS.     lv_s_pri_clr = 'bg-slate-100 text-slate-600'.
    ENDCASE.

    " Kategori color
    CASE ls_mat_hist-kategori.
      WHEN 'DRW'. lv_s_kat_clr = 'bg-emerald-100 text-emerald-700 border-emerald-200'.
      WHEN 'DBX'. lv_s_kat_clr = 'bg-blue-100 text-blue-700 border-blue-200'.
      WHEN 'DMT'. lv_s_kat_clr = 'bg-orange-100 text-orange-700 border-orange-200'.
      WHEN OTHERS. lv_s_kat_clr = 'bg-slate-100 text-slate-600'.
    ENDCASE.

    " Date format
    DATA: lv_cd TYPE string.
    lv_cd = |{ ls_mat_hist-created_date+6(2) }.{ ls_mat_hist-created_date+4(2) }.{ ls_mat_hist-created_date(4) }|.

    " Cost
    IF ls_mat_hist-cost_estimate IS NOT INITIAL.
      DATA: lv_cost_str TYPE string.
      lv_cost_str = ls_mat_hist-cost_estimate.
      lv_s_cost = |Rp { lv_cost_str }|.
    ELSE.
      lv_s_cost = '-'.
    ENDIF.

    " DMS link HTML (data attribute + JS click)
    IF ls_mat_hist-dir_number_clean IS NOT INITIAL.
      CONCATENATE '<div class="dms-link flex items-center gap-1.5'
        ' cursor-pointer hover:bg-green-50'
        ' px-2 py-1 rounded transition-colors'
        ' inline-flex"'
        ' data-dmsnr="' ls_mat_hist-dir_number_clean '">'
        '<span class="w-2 h-2 rounded-full'
        ' bg-green-500"></span>'
        '<span class="text-xs font-semibold'
        ' text-green-700">'
        ls_mat_hist-dir_number_clean
        '</span></div>'
        INTO lv_s_dms.
    ELSE.
      lv_s_dms = '<span class="text-xs text-slate-400">-</span>'.
    ENDIF.

    " Reject reason
    IF ls_mat_hist-reject_reason IS NOT INITIAL AND ls_mat_hist-status = 'REJC'.
      CONCATENATE '<div class="mt-2 px-3 py-2 bg-red-50'
        ' border-l-[3px] border-red-400 rounded'
        ' text-xs text-red-700">'
        '<strong>Alasan Reject:</strong> '
        ls_mat_hist-reject_reason '</div>'
        INTO lv_s_reject.
    ELSE.
      CLEAR lv_s_reject.
    ENDIF.

    " Escape single quotes for HTML attributes
    DATA: lv_esc_title TYPE string,
          lv_esc_descr TYPE string.
    lv_esc_title = ls_mat_hist-title.
    lv_esc_descr = ls_mat_hist-descr.

    " Build card HTML (green theme)
    CONCATENATE lv_mat_html
      '<div class="bg-white rounded-xl border-l-4 border-l-emerald-500'
      ' border border-slate-200 shadow-sm overflow-hidden'
      ' hover:shadow-md transition-shadow mb-4">'
      '<div class="px-5 py-3 bg-gradient-to-r from-emerald-50'
      ' to-white border-b border-slate-100'
      ' flex items-center justify-between">'
      '  <div class="flex items-center gap-2 flex-wrap">'
      '    <span class="text-sm font-extrabold text-emerald-800'
      ' font-mono"><i class="fas fa-file-circle-exclamation'
      ' text-emerald-500 mr-1"></i>' ls_mat_hist-ecr_no '</span>'
      '    <span class="px-2.5 py-0.5 rounded-full text-[0.6rem]'
      ' font-bold ' lv_s_badge '">'
      lv_s_status '</span>'

  " ================================================================

    SELECT a~ecr_no, a~title, a~matnr_fg, a~descr, a~kategori, a~priority,
           a~requester, a~drafter, a~dokar, a~dir_number, a~status,

           c~zecm_full_name AS requester_name,

           b~maktx

      FROM zecm_req AS a
      LEFT OUTER JOIN makt AS b
        ON a~matnr_fg = b~matnr AND b~spras = @sy-langu
      LEFT OUTER JOIN zecm_user AS c
        ON a~requester = c~zecm_user_id
      LEFT OUTER JOIN zecm_user AS d
        ON a~drafter = d~zecm_user_id
      ORDER BY a~created_date DESCENDING, a~created_time DESCENDING.

  " Process each ECR
  DATA: lv_matnr_conv_a TYPE matnr,
        lv_doknr_a TYPE doknr,
        lv_dokar_a TYPE dokar,
        lv_dokvr_a TYPE char2,
        lv_doktl_a TYPE char3,
        lv_dokst_a TYPE char2,
        lv_objky_a TYPE char90,
        lv_matnr_raw_a TYPE char90,
        lv_objky_clean_a TYPE char90,
        lv_dir_parts TYPE STANDARD TABLE OF string,
        lv_dir_doknr TYPE string.

  DATA: lv_target_dokst TYPE char2.


  LOOP AT LT_ECR INTO LS_ECR.
    " Remove leading zeros from ECR number
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
      EXPORTING  input  = LS_ECR-ecr_no
      IMPORTING  output = LS_ECR-ecr_no.

    " Material no zero
    LS_ECR-matnr_fg_no_zero = LS_ECR-matnr_fg.
    SHIFT LS_ECR-matnr_fg_no_zero LEFT DELETING LEADING '0'.
    
    " Fetch description with ALPHA_INPUT if JOIN failed
    IF LS_ECR-maktx IS INITIAL.
      DATA: lv_fix_matnr TYPE matnr.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
        EXPORTING input = LS_ECR-matnr_fg
        IMPORTING output = lv_fix_matnr.
      SELECT SINGLE maktx FROM makt INTO @LS_ECR-maktx
        WHERE matnr = @lv_fix_matnr AND spras = @sy-langu.
    ENDIF.

    " Clean Title Prefix
    DATA: lv_ecr_title_clean TYPE string.
    lv_ecr_title_clean = LS_ECR-title.
    IF lv_ecr_title_clean CS 'Trial Konstruksi Groove / Kerf "'.
      REPLACE FIRST OCCURRENCE OF 'Trial Konstruksi Groove / Kerf "' IN lv_ecr_title_clean WITH ''.
      REPLACE ALL OCCURRENCES OF '"' IN lv_ecr_title_clean WITH ''.
    ELSEIF lv_ecr_title_clean CS 'Trial Konstruksi Groove / Kerf Cut "'.
      REPLACE FIRST OCCURRENCE OF 'Trial Konstruksi Groove / Kerf Cut "' IN lv_ecr_title_clean WITH ''.
      REPLACE ALL OCCURRENCES OF '"' IN lv_ecr_title_clean WITH ''.
    ELSEIF lv_ecr_title_clean CS 'Trial Konstruksi Groove/Kerf "'.
      REPLACE FIRST OCCURRENCE OF 'Trial Konstruksi Groove/Kerf "' IN lv_ecr_title_clean WITH ''.
      REPLACE ALL OCCURRENCES OF '"' IN lv_ecr_title_clean WITH ''.
    ENDIF.
    LS_ECR-title = lv_ecr_title_clean.


    " Clean dir_number
    IF LS_ECR-dir_number IS NOT INITIAL.
      SPLIT LS_ECR-dir_number AT '/' INTO TABLE lv_dir_parts.
      READ TABLE lv_dir_parts INTO lv_dir_doknr INDEX 2.
      IF lv_dir_doknr IS NOT INITIAL.
        SHIFT lv_dir_doknr LEFT DELETING LEADING '0'.
        LS_ECR-dir_number_clean = lv_dir_doknr.
      ELSE.
        LS_ECR-dir_number_clean = LS_ECR-dir_number.
      ENDIF.
    ENDIF.

    " Fast DMS detection (Single Select, no table scan)
    CLEAR: LS_ECR-dms_doknr, LS_ECR-dms_dokar, LS_ECR-dms_dokvr, LS_ECR-dms_doktl, LS_ECR-dms_dokst, LS_ECR-dms_found.

    IF LS_ECR-dokar IS NOT INITIAL AND LS_ECR-dir_number IS NOT INITIAL.
      SELECT SINGLE doknr, dokar, dokvr, doktl, dokst
        INTO (@LS_ECR-dms_doknr, @LS_ECR-dms_dokar, @LS_ECR-dms_dokvr, @LS_ECR-dms_doktl, @LS_ECR-dms_dokst)
        FROM draw
        WHERE dokar = @LS_ECR-dokar AND doknr = @LS_ECR-dir_number.
      IF sy-subrc = 0.
        LS_ECR-dms_found = abap_true.
      ENDIF.
    ENDIF.

    IF LS_ECR-dms_found IS INITIAL AND LS_ECR-matnr_fg IS NOT INITIAL.
      lv_matnr_conv_a = LS_ECR-matnr_fg.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
        EXPORTING  input  = lv_matnr_conv_a
        IMPORTING  output = lv_matnr_conv_a.
      SELECT SINGLE dokar, doknr, dokvr, doktl
        INTO (@LS_ECR-dms_dokar, @LS_ECR-dms_doknr, @LS_ECR-dms_dokvr, @LS_ECR-dms_doktl)
        FROM drad
        WHERE dokob = 'MARA' AND objky = @lv_matnr_conv_a.
      IF sy-subrc = 0.
        LS_ECR-dms_found = abap_true.
      ENDIF.
    ENDIF.

    IF LS_ECR-dms_doknr IS NOT INITIAL.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
        EXPORTING  input  = LS_ECR-dms_doknr
        IMPORTING  output = LS_ECR-dms_doknr.
    ENDIF.

    MODIFY LT_ECR FROM LS_ECR.
  ENDLOOP.
  " Sort LT_ECR: CRTD (Draft) first, then INSP, then by date (already sorted by date from DB)
  DATA: lt_ecr_temp LIKE LT_ECR.

  SELECT zecm_log~log_id, zecm_log~ecr_no, zecm_log~action, zecm_log~old_status, zecm_log~new_status, zecm_log~user_id,
         e~zecm_full_name AS user_name, zecm_log~action_date, zecm_log~action_time, zecm_log~reason
         INTO TABLE @LT_LOGS
         FROM zecm_log
         LEFT OUTER JOIN zecm_user AS e
           ON zecm_log~user_id = e~zecm_user_id
         WHERE zecm_log~ecr_no = @GV_SELECTED_IREQ_AUDIT
         AND zecm_log~is_ireq = 'X'
         ORDER BY zecm_log~action_date DESCENDING, zecm_log~action_time DESCENDING.

  LOOP AT LT_LOGS INTO LS_LOG.
    CASE LS_LOG-old_status.
      WHEN 'REQ'.  LS_LOG-old_label = 'Pending Approval'.

      WHEN 'ECO'.  LS_LOG-old_label = 'ECO Created'.
      WHEN 'REJC'. LS_LOG-old_label = 'Rejected'.
      WHEN 'PROC'. LS_LOG-old_label = 'In Process'.
      WHEN OTHERS. LS_LOG-old_label = LS_LOG-old_status.
    ENDCASE.
    CASE LS_LOG-new_status.
      WHEN 'REQ'.  LS_LOG-new_label = 'Pending Approval'.
      WHEN 'APPR'. LS_LOG-new_label = 'Approved'.
      WHEN 'ECO'.  LS_LOG-new_label = 'ECO Created'.

      WHEN OTHERS. LS_LOG-new_label = LS_LOG-new_status.
