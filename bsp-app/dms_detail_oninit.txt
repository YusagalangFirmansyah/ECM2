* dms_detail OnInit
GV_USER_ID   = request->get_form_field( 'u' ).
GV_FULL_NAME = request->get_form_field( 'n' ).
GV_ROLE      = request->get_form_field( 'r' ).
MV_DOKAR     = request->get_form_field( 'DOKAR' ).
MV_DOKNR     = request->get_form_field( 'DOKNR' ).
MV_DOKVR     = request->get_form_field( 'DOKVR' ).
MV_DOKTL     = request->get_form_field( 'DOKTL' ).

IF MV_DOKNR IS INITIAL.
  MV_ERROR = 'Parameter DOKNR tidak valid.'.
  RETURN.
ENDIF.

" Format DOKNR dengan 25 digit leading zeros
CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
  EXPORTING  input  = MV_DOKNR
  IMPORTING  output = MV_DOKNR.

" If DOKAR is empty, lookup from DRAW by DOKNR
IF MV_DOKAR IS INITIAL.
  SELECT dokar, dokvr, doktl
    INTO (@MV_DOKAR, @MV_DOKVR, @MV_DOKTL)
    FROM draw
    WHERE doknr = @MV_DOKNR
    ORDER BY dokvr DESCENDING, doktl DESCENDING.
    EXIT.
  ENDSELECT.
ENDIF.

" Pastikan record DRAW eksis. Jika DOKVR / DOKTL spesifik tidak ada di DRAW,
" ambil versi terbaru yang valid dari DRAW
DATA: lv_draw_exists TYPE abap_bool.
IF MV_DOKVR IS NOT INITIAL AND MV_DOKTL IS NOT INITIAL.
  SELECT SINGLE @abap_true FROM draw INTO @lv_draw_exists
    WHERE dokar = @MV_DOKAR
      AND doknr = @MV_DOKNR
      AND dokvr = @MV_DOKVR
      AND doktl = @MV_DOKTL.
ENDIF.

IF lv_draw_exists <> abap_true.
  " Prioritaskan versi aktif (IA / AP / FR) terlebih dahulu sebelum draft AA
  SELECT dokvr, doktl
    INTO (@MV_DOKVR, @MV_DOKTL)
    FROM draw
    WHERE dokar = @MV_DOKAR
      AND doknr = @MV_DOKNR
      AND dokst IN ('IA', 'AP', 'FR')
    ORDER BY dokvr DESCENDING, doktl DESCENDING.
    EXIT.
  ENDSELECT.

  IF MV_DOKVR IS INITIAL.
    SELECT dokvr, doktl
      INTO (@MV_DOKVR, @MV_DOKTL)
      FROM draw
      WHERE dokar = @MV_DOKAR
        AND doknr = @MV_DOKNR
      ORDER BY dokvr DESCENDING, doktl DESCENDING.
      EXIT.
    ENDSELECT.
  ENDIF.
ENDIF.

IF MV_DOKTL IS INITIAL.
  MV_DOKTL = '000'.
ENDIF.
IF MV_DOKVR IS INITIAL.
  MV_DOKVR = '00'.
ENDIF.

" 1. Load document header + files via CVAPI_DOC_GETDETAIL
DATA: ls_draw    TYPE draw,
      lt_files   TYPE STANDARD TABLE OF cvapi_doc_file.

CALL FUNCTION 'CVAPI_DOC_GETDETAIL'
  EXPORTING
    pf_batchmode    = 'X'
    pf_dokar        = MV_DOKAR
    pf_doknr        = MV_DOKNR
    pf_dokvr        = MV_DOKVR
    pf_doktl        = MV_DOKTL
    pf_active_files = 'X'
  IMPORTING
    psx_draw        = ls_draw
  TABLES
    pt_files        = lt_files
  EXCEPTIONS
    OTHERS          = 4.

" Fallback 1: Coba CVAPI tanpa pf_active_files jika kosong
IF lt_files IS INITIAL.
  CALL FUNCTION 'CVAPI_DOC_GETDETAIL'
    EXPORTING
      pf_batchmode    = 'X'
      pf_dokar        = MV_DOKAR
      pf_doknr        = MV_DOKNR
      pf_dokvr        = MV_DOKVR
      pf_doktl        = MV_DOKTL
      pf_active_files = space
    IMPORTING
      psx_draw        = ls_draw
    TABLES
      pt_files        = lt_files
    EXCEPTIONS
      OTHERS          = 4.
ENDIF.

mt_files = lt_files.

" 2. Load header info via BAPI_DOCUMENT_GETDETAIL2
DATA: ls_header   TYPE bapi_doc_draw2,
      ls_ret_bapi TYPE bapiret2,
      lt_bapi_fls TYPE STANDARD TABLE OF bapi_doc_files2,
      ls_bapi_fl  TYPE bapi_doc_files2,
      ls_conv_fl  TYPE cvapi_doc_file.

CALL FUNCTION 'BAPI_DOCUMENT_GETDETAIL2'
  EXPORTING
    documenttype         = MV_DOKAR
    documentnumber       = MV_DOKNR
    documentpart         = MV_DOKTL
    documentversion      = MV_DOKVR
    getdocfiles          = 'X'
    getactivefiles       = 'X'
  IMPORTING
    documentdata         = ls_header
    return               = ls_ret_bapi
  TABLES
    documentfiles        = lt_bapi_fls.

ms_header = ls_header.

" Fallback Header: Jika BAPI return kosong, isi dari DRAW
IF ms_header-documentnumber IS INITIAL.
  ms_header-documenttype    = MV_DOKAR.
  ms_header-documentnumber  = MV_DOKNR.
  ms_header-documentversion = MV_DOKVR.
  ms_header-documentpart    = MV_DOKTL.
  ms_header-statusintern    = ls_draw-dokst.
  IF ls_draw-dwnam IS NOT INITIAL.
    ms_header-username      = ls_draw-dwnam.
  ELSE.
    ms_header-username      = sy-uname.
  ENDIF.
  IF ls_draw-adatum IS NOT INITIAL.
    ms_header-createdate    = ls_draw-adatum.
  ELSE.
    ms_header-createdate    = sy-datum.
  ENDIF.
ENDIF.

IF ms_header-description IS INITIAL.
  SELECT SINGLE dktxt FROM drat INTO @ms_header-description
    WHERE dokar = @MV_DOKAR
      AND doknr = @MV_DOKNR
      AND dokvr = @MV_DOKVR
      AND doktl = @MV_DOKTL.
ENDIF.

" Fallback 2: Jika CVAPI kosong tetapi BAPI mendeteksi file
IF mt_files IS INITIAL AND lt_bapi_fls IS NOT INITIAL.
  LOOP AT lt_bapi_fls INTO ls_bapi_fl.
    CLEAR ls_conv_fl.
    ls_conv_fl-filename    = ls_bapi_fl-docfile.
    ls_conv_fl-description = ls_bapi_fl-description.
    ls_conv_fl-storage_cat = ls_bapi_fl-storagecategory.
    ls_conv_fl-dappl       = ls_bapi_fl-wsapplication.
    APPEND ls_conv_fl TO mt_files.
  ENDLOOP.
ENDIF.

" Fallback 3: Baca langsung dari tabel transparan DMS_DOC_FILES
IF mt_files IS INITIAL.
  DATA: lt_db_files TYPE STANDARD TABLE OF dms_doc_files,
        ls_db_file  TYPE dms_doc_files.
  SELECT * FROM dms_doc_files INTO TABLE lt_db_files
    WHERE dokar = MV_DOKAR
      AND doknr = MV_DOKNR
      AND dokvr = MV_DOKVR
      AND doktl = MV_DOKTL
    ORDER BY file_idx.
  LOOP AT lt_db_files INTO ls_db_file.
    CLEAR ls_conv_fl.
    ls_conv_fl-filename = ls_db_file-filename.
    ls_conv_fl-dappl    = ls_db_file-dappl.
    APPEND ls_conv_fl TO mt_files.
  ENDLOOP.
ENDIF.

" Fallback 4: File klasik disimpan di header DRAW (FILEP)
IF mt_files IS INITIAL AND ls_draw-filep IS NOT INITIAL.
  CLEAR ls_conv_fl.
  ls_conv_fl-filename = ls_draw-filep.
  ls_conv_fl-dappl    = ls_draw-dappl.
  APPEND ls_conv_fl TO mt_files.
ENDIF.

" Load Status Log dari DRAP
DATA: lt_status_log TYPE STANDARD TABLE OF drap.
SELECT * FROM drap INTO TABLE lt_status_log
  WHERE dokar = MV_DOKAR
    AND doknr = MV_DOKNR
    AND dokvr = MV_DOKVR
    AND doktl = MV_DOKTL
  ORDER BY stzae DESCENDING.
mt_status_log = lt_status_log.

" Load Object Links (Material) dari DRAD + MAKT
DATA: ls_obj_link TYPE ty_obj_link.
SELECT c~objky, d~maktx, e~mtart
  INTO TABLE @mt_obj_links
  FROM drad AS c
  LEFT JOIN makt AS d
    ON c~objky = d~matnr
   AND d~spras = @sy-langu
  LEFT JOIN mara AS e
    ON c~objky = e~matnr
  WHERE c~dokar = @MV_DOKAR
    AND c~doknr = @MV_DOKNR
    AND c~dokvr = @MV_DOKVR
    AND c~doktl = @MV_DOKTL
  ORDER BY c~objky.

" Remove leading zeros from material numbers
LOOP AT mt_obj_links INTO ls_obj_link.
  CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
    EXPORTING  input  = ls_obj_link-objky
    IMPORTING  output = ls_obj_link-objky.
  MODIFY mt_obj_links FROM ls_obj_link.
ENDLOOP.

" Assign sort priority: FERT=1, HALB=2, HALM=3, others=4, empty=5
LOOP AT mt_obj_links INTO ls_obj_link.
  CASE ls_obj_link-mtart.
    WHEN 'FERT'. ls_obj_link-sort_priority = 1.
    WHEN 'HALB'. ls_obj_link-sort_priority = 2.
    WHEN 'HALM'. ls_obj_link-sort_priority = 3.
    WHEN ''  .   ls_obj_link-sort_priority = 5.
    WHEN OTHERS. ls_obj_link-sort_priority = 4.
  ENDCASE.
  MODIFY mt_obj_links FROM ls_obj_link.
ENDLOOP.

" Sort: priority first, then material number
SORT mt_obj_links BY sort_priority objky.

" Limit to 5
IF lines( mt_obj_links ) > 5.
  DELETE mt_obj_links FROM 6.
ENDIF.

" Set Navigation Parameters
navigation->set_parameter( name = 'u' value = GV_USER_ID ).
navigation->set_parameter( name = 'n' value = GV_FULL_NAME ).
navigation->set_parameter( name = 'r' value = GV_ROLE ).
