" drafter_list Event Handler â€” V08_e
" + upload_bom + download_bom actions for BOM Gate

DATA: lv_action      TYPE string,
      lv_ecr_no      TYPE numc10,
      lv_dokar       TYPE dokar,
      lv_old_status  TYPE zecm_req-status,
      lv_url         TYPE string,
      lv_matnr_conv  TYPE matnr,
      lv_doknr       TYPE doknr,
      lv_dokvr       TYPE dokvr,
      lv_doktl       TYPE doktl_d,
      lv_dokst       TYPE dokst,
      lv_dir_number  TYPE zecm_req-dir_number,
      ls_log         TYPE zecm_log,
      ls_doc_data    TYPE bapi_doc_draw2,
      ls_doc_datax   TYPE bapi_doc_drawx2,
      ls_return      TYPE bapiret2.

lv_action = request->get_form_field( 'OnInputProcessing' ).

" ============================================================
" AUTHORIZATION CHECK (Security Guard)
" Pastikan Drafter hanya bisa modifikasi ECR miliknya sendiri
" ============================================================
IF lv_action = 'upload_bom' OR lv_action = 'submit_ecr' OR lv_action = 'skip_bom' OR 
   lv_action = 'cancel_upload_bom' OR lv_action = 'reject_ecr' OR lv_action = 'cancel_ecr'.
  
  DATA: lv_auth_ecr TYPE numc10.
  lv_auth_ecr = request->get_form_field( 'ecr_no' ).
  IF lv_auth_ecr IS NOT INITIAL.
    DATA: lv_auth_drafter TYPE zecm_req-drafter.
    SELECT SINGLE drafter FROM zecm_req INTO @lv_auth_drafter WHERE ecr_no = @lv_auth_ecr.
    IF sy-subrc = 0 AND lv_auth_drafter <> GV_USER_ID.
      GV_MESSAGE = 'Akses Ditolak: Anda bukan PIC Drafter untuk ECR ini.'.
      GV_ICON    = 'error'.
      CONCATENATE 'drafter_list.htm?u=' GV_USER_ID '&r=' GV_ROLE '&n=' GV_FULL_NAME '&msg=' GV_MESSAGE '&icon=' GV_ICON INTO lv_url.
      navigation->goto_page( lv_url ).
      RETURN.
    ENDIF.
  ENDIF.
ENDIF.

" ============================================================
" UPLOAD BOM ATTACHMENT (multipart binary â†’ ZECM_BOM_ATT)
" ============================================================
IF lv_action = 'upload_bom'.
  lv_ecr_no = request->get_form_field( 'ecr_no' ).
  GV_BOM_NOTES = request->get_form_field( 'bom_notes' ).

  IF lv_ecr_no IS INITIAL.
    GV_MESSAGE = 'ECR tidak valid.'.
    GV_ICON    = 'error'.
    CONCATENATE 'drafter_list.htm?u=' GV_USER_ID '&r=' GV_ROLE '&n=' GV_FULL_NAME '&msg=' GV_MESSAGE '&icon=' GV_ICON INTO lv_url.
    navigation->goto_page( lv_url ). RETURN.
  ENDIF.

  " Check ECR status â€” must be INWK to upload BOM
  DATA: lv_bom_status TYPE zecm_req-status.
  SELECT SINGLE status FROM zecm_req INTO @lv_bom_status
    WHERE ecr_no = @lv_ecr_no.
  IF lv_bom_status <> 'INWK'.
    GV_MESSAGE = 'Hanya ECR dengan status In Work yang bisa upload BOM.'.
    GV_ICON    = 'error'.
    CONCATENATE 'drafter_list.htm?u=' GV_USER_ID '&r=' GV_ROLE '&n=' GV_FULL_NAME '&msg=' GV_MESSAGE '&icon=' GV_ICON INTO lv_url.
    navigation->goto_page( lv_url ). RETURN.
  ENDIF.

  " Parse Base64 form data (sent via JS FileReader)
  DATA: lv_b64       TYPE string,
        lv_file_name TYPE string,
        lv_mime_type TYPE string,
        lv_file_ext  TYPE string,
        lv_file_bin  TYPE xstring,
        lv_att_count TYPE i VALUE 0.

  lv_b64       = request->get_form_field( 'bom_file_b64' ).
  lv_file_name = request->get_form_field( 'bom_file_name' ).
  lv_mime_type = request->get_form_field( 'bom_file_type' ).

  IF lv_b64 IS INITIAL OR lv_file_name IS INITIAL.
    GV_MESSAGE = 'File BOM wajib diisi.'.
    GV_ICON    = 'error'.
    CONCATENATE 'drafter_list.htm?u=' GV_USER_ID '&r=' GV_ROLE '&n=' GV_FULL_NAME '&msg=' GV_MESSAGE '&icon=' GV_ICON INTO lv_url.
    navigation->goto_page( lv_url ). RETURN.
  ENDIF.

  " Clean and Decode Base64 to XSTRING
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>cr_lf IN lv_b64 WITH ''.
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>newline IN lv_b64 WITH ''.
  TRANSLATE lv_b64 USING ' +'.

  CALL FUNCTION 'SSFC_BASE64_DECODE'
    EXPORTING
      b64data                  = lv_b64
    IMPORTING
      bindata                  = lv_file_bin
    EXCEPTIONS
      ssfc_fault               = 1
      ssfc_base64_decode_error = 2
      OTHERS                   = 3.

  IF sy-subrc <> 0 OR lv_file_bin IS INITIAL.
    GV_MESSAGE = 'Gagal memproses file.'.
    GV_ICON    = 'error'.
    CONCATENATE 'drafter_list.htm?u=' GV_USER_ID '&r=' GV_ROLE '&n=' GV_FULL_NAME '&msg=' GV_MESSAGE '&icon=' GV_ICON INTO lv_url.
    navigation->goto_page( lv_url ). RETURN.
  ENDIF.

  " Extract extension & validate PDF safely (supports multiple dots e.g. BOM_rev.01.pdf)
  DATA: lt_fn_parts TYPE TABLE OF string,
        lv_fn_lines TYPE i,
        lv_is_excel TYPE c LENGTH 1 VALUE '',
        lv_fn_upper TYPE string,
        lv_mime_low TYPE string.

  " 1. Cek ekstensi dari titik paling akhir
  SPLIT lv_file_name AT '.' INTO TABLE lt_fn_parts.
  DESCRIBE TABLE lt_fn_parts LINES lv_fn_lines.
  IF lv_fn_lines > 1.
    READ TABLE lt_fn_parts INDEX lv_fn_lines INTO lv_file_ext.
    CONDENSE lv_file_ext NO-GAPS.
    TRANSLATE lv_file_ext TO LOWER CASE.
    IF lv_file_ext = 'xls' OR lv_file_ext = 'xlsx'.
      lv_is_excel = 'X'.
    ENDIF.
  ENDIF.

  " 2. Fallback: cek jika nama file berakhiran atau mengandung .PDF
  IF lv_is_excel IS INITIAL.
    lv_fn_upper = lv_file_name.
    TRANSLATE lv_fn_upper TO UPPER CASE.
    IF lv_fn_upper CS '.XLS'.
      lv_is_excel   = 'X'.
    ENDIF.
  ENDIF.

  " 3. Fallback: cek MIME Type dari browser
  IF lv_is_excel IS INITIAL.
    lv_mime_low = lv_mime_type.
    TRANSLATE lv_mime_low TO LOWER CASE.
    IF lv_mime_low CS 'excel' OR lv_mime_low CS 'spreadsheet'.
      lv_is_excel   = 'X'.
    ENDIF.
  ENDIF.

  IF lv_is_excel IS INITIAL.
    GV_MESSAGE = 'Hanya file Excel yang diizinkan.'.
    GV_ICON    = 'error'.
    CONCATENATE 'drafter_list.htm?u=' GV_USER_ID '&r=' GV_ROLE '&n=' GV_FULL_NAME '&msg=' GV_MESSAGE '&icon=' GV_ICON INTO lv_url.
    navigation->goto_page( lv_url ). RETURN.
  ENDIF.

  " Get next ATT_ID
  DATA: lv_max_att_id TYPE zecm_bom_att-att_id.
  SELECT MAX( att_id ) FROM zecm_bom_att INTO @lv_max_att_id
    WHERE ecr_no = @lv_ecr_no.

  lv_max_att_id = lv_max_att_id + 1.

  " Insert to ZECM_BOM_ATT
  DATA: ls_bom_att TYPE zecm_bom_att.
  CLEAR ls_bom_att.
  ls_bom_att-mandt        = sy-mandt.
  ls_bom_att-ecr_no       = lv_ecr_no.
  ls_bom_att-att_id       = lv_max_att_id.
  ls_bom_att-file_name    = lv_file_name.
  ls_bom_att-file_ext     = lv_file_ext.
  ls_bom_att-mime_type    = lv_mime_type.
  ls_bom_att-file_size    = xstrlen( lv_file_bin ).
  ls_bom_att-file_data    = lv_file_bin.
  ls_bom_att-uploaded_by  = GV_USER_ID.
  ls_bom_att-uploaded_at  = sy-datum.
  ls_bom_att-uploaded_time = sy-uzeit.

  INSERT zecm_bom_att FROM ls_bom_att.

  IF sy-subrc = 0.
    lv_att_count = 1.
  ENDIF.

  " Update BOM_CHECK = 'P' and status = 'BMPR' if files uploaded
  IF lv_att_count > 0.
    UPDATE zecm_req SET
      status       = 'BMPR',
      bom_check    = 'P',
      bom_notes    = @GV_BOM_NOTES,
      changed_date = @sy-datum,
      changed_time = @sy-uzeit
    WHERE ecr_no = @lv_ecr_no.

    " Audit log
    CALL FUNCTION 'NUMBER_GET_NEXT'
      EXPORTING nr_range_nr = '01' object = 'ZECM_LOG'
      IMPORTING number = ls_log-log_id
      EXCEPTIONS OTHERS = 1.

    IF sy-subrc = 0.
      ls_log-mandt       = sy-mandt.
      ls_log-ecr_no      = lv_ecr_no.
      ls_log-action      = 'BOM_PENDING'.
      ls_log-old_status  = lv_bom_status.
      ls_log-new_status  = 'BMPR'.
      ls_log-user_id     = GV_USER_ID.
      ls_log-action_date = sy-datum.
      ls_log-action_time = sy-uzeit.
      ls_log-reason      = '1 file Excel uploaded'.
      INSERT zecm_log FROM ls_log.
    ENDIF.

    COMMIT WORK AND WAIT.

    GV_MESSAGE = 'File BOM berhasil di-upload. Menunggu review Master Data.'.
    GV_ICON    = 'success'.
  ELSE.
    GV_MESSAGE = 'Gagal menyimpan file ke database.'.
    GV_ICON    = 'error'.
  ENDIF.

  CONCATENATE 'drafter_list.htm?u=' GV_USER_ID '&r=' GV_ROLE '&n=' GV_FULL_NAME '&msg=' GV_MESSAGE '&icon=' GV_ICON INTO lv_url.
  navigation->goto_page( lv_url ). RETURN.
ENDIF.

" DOWNLOAD BOM ATTACHMENT (binary response from ZECM_BOM_ATT)
" ============================================================
IF lv_action = 'download_bom'.
  DATA: lv_dl_ecr_no TYPE zecm_bom_att-ecr_no,
        lv_dl_att_id TYPE zecm_bom_att-att_id,
        ls_dl_att    TYPE zecm_bom_att,
        lv_mime_str  TYPE string,
        lv_fname_str TYPE string.

  lv_dl_ecr_no = request->get_form_field( 'ecr_no' ).
  lv_dl_att_id = request->get_form_field( 'att_id' ).

  SELECT SINGLE * FROM zecm_bom_att INTO @ls_dl_att
    WHERE ecr_no = @lv_dl_ecr_no AND att_id = @lv_dl_att_id.

  IF sy-subrc = 0.
    lv_mime_str = ls_dl_att-mime_type.
    lv_fname_str = ls_dl_att-file_name.
    runtime->server->response->set_header_field(
      name  = 'Content-Type'
      value = lv_mime_str ).
    runtime->server->response->set_header_field(
      name  = 'Content-Disposition'
      value = |attachment; filename="{ lv_fname_str }"| ).
    runtime->server->response->set_data( data = ls_dl_att-file_data ).
    RETURN.
  ELSE.
    GV_MESSAGE = 'File tidak ditemukan.'.
    GV_ICON    = 'error'.
    CONCATENATE 'drafter_list.htm?u=' GV_USER_ID '&r=' GV_ROLE '&n=' GV_FULL_NAME '&msg=' GV_MESSAGE '&icon=' GV_ICON INTO lv_url.
    navigation->goto_page( lv_url ). RETURN.
  ENDIF.
ENDIF.

" ============================================================
" SEARCH MATERIAL (from modal â€” keep modal open)
" ============================================================
IF lv_action = 'search_material'.
  LV_MAT_KEYWORD = request->get_form_field( 'keyword' ).
  LV_MODAL_OPEN = 'X'.

  IF LV_MAT_KEYWORD IS NOT INITIAL.
    DATA: lv_q TYPE matnr, lv_qs TYPE string, lv_html TYPE string.
    DATA: BEGIN OF ls_m,
            matnr TYPE mara-matnr,
            maktx TYPE makt-maktx,
            mtart TYPE mara-mtart,
          END OF ls_m,
          lt_m LIKE STANDARD TABLE OF ls_m.

    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING input  = LV_MAT_KEYWORD
      IMPORTING output = lv_q.
    lv_qs = LV_MAT_KEYWORD.
    TRANSLATE lv_qs TO UPPER CASE.
    CONCATENATE '%' lv_qs '%' INTO lv_qs.

    SELECT a~matnr, b~maktx, a~mtart
      INTO TABLE @lt_m
      FROM mara AS a
      INNER JOIN makt AS b ON b~matnr = a~matnr AND b~spras = @sy-langu
      WHERE ( a~matnr = @lv_q
         OR a~matnr LIKE @lv_qs
         OR b~maktx LIKE @lv_qs )
        AND a~mtart IN ( 'FERT', 'HALB', 'HALM' )
      ORDER BY a~mtart ASCENDING, a~matnr ASCENDING.

    CLEAR lv_html.
    DATA: lv_mnr TYPE string,
          lv_mtart_bg TYPE string,
          lv_mtart_s TYPE string,
          lv_maktx_safe TYPE string.
    LOOP AT lt_m INTO ls_m.
      lv_mnr = ls_m-matnr.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
        EXPORTING input  = lv_mnr
        IMPORTING output = lv_mnr.

      " HTML-encode maktx for safe use in data-* attributes
      lv_maktx_safe = ls_m-maktx.
      REPLACE ALL OCCURRENCES OF '&' IN lv_maktx_safe WITH '&amp;'.
      REPLACE ALL OCCURRENCES OF '"' IN lv_maktx_safe WITH '&quot;'.

      CLEAR lv_mtart_bg.
      CASE ls_m-mtart.
        WHEN 'FERT'. lv_mtart_bg = 'bg-emerald-100 text-emerald-700'. lv_mtart_s = 'Finish Goods'.
        WHEN 'HALB'. lv_mtart_bg = 'bg-blue-100 text-blue-700'.       lv_mtart_s = 'HALB'.
        WHEN 'HALM'. lv_mtart_bg = 'bg-orange-100 text-orange-700'.   lv_mtart_s = 'HALM'.
        WHEN OTHERS. lv_mtart_bg = 'bg-slate-100 text-slate-700'.     lv_mtart_s = ls_m-mtart.
      ENDCASE.

      CONCATENATE lv_html
        '<div class="flex items-center justify-between px-3 py-2 hover:bg-emerald-50 cursor-pointer border-b border-slate-100 last:border-0 mat-pick"'
        ' data-matnr="' ls_m-matnr '"'
        ' data-maktx="' lv_maktx_safe '">'
        '<div>'
        '<span class="text-[0.65rem] px-2 py-0.5 rounded font-extrabold tracking-wide ' lv_mtart_bg '">' lv_mtart_s '</span>'
        ' <span class="text-xs text-slate-700 ml-1">' lv_mnr '</span>'
        '<div class="text-xs text-slate-600 mt-0.5">' ls_m-maktx '</div>'
        '</div>'
        '<span class="text-emerald-600 text-xs font-bold">+ Pilih</span></div>'
        INTO lv_html.
    ENDLOOP.

    IF lv_html IS INITIAL.
      GV_MAT_RESULTS = '<div class="text-center py-3 text-slate-400 text-xs">Tidak ditemukan</div>'.
    ELSE.
      GV_MAT_RESULTS = lv_html.
    ENDIF.
  ENDIF.

  " DON'T redirect â€” just fall through to OnInit + Layout
  RETURN.
ENDIF.

" CREATE ECR (from modal â€” drafter creates new ECR)
IF lv_action = 'create_ecr'.
  DATA: lv_title     TYPE string,
        lv_desc      TYPE string,
        lv_kat       TYPE string,
        lv_prio      TYPE string,
        lv_cost      TYPE string,
        lv_dept      TYPE string,
        lv_reason    TYPE string,
        lv_matnr_str TYPE string,
        lv_maktx     TYPE makt-maktx,
        ls_req       TYPE zecm_req,
        lv_count     TYPE i VALUE 0.

  lv_title     = request->get_form_field( 'title' ).
  lv_desc      = request->get_form_field( 'description' ).
  lv_kat       = request->get_form_field( 'kategori' ).
  lv_prio      = request->get_form_field( 'priority' ).
  lv_cost      = request->get_form_field( 'cost_estimate' ).
  lv_dept      = request->get_form_field( 'dept' ).
  lv_reason    = request->get_form_field( 'reason' ).
  lv_matnr_str = request->get_form_field( 'materials' ).

  " Security: Validate kategori per drafter role
  IF GV_ROLE = 'DRAFTER_WOOD' AND lv_kat <> 'DRW' AND lv_kat <> 'DBX'.
    GV_MESSAGE = 'Role DRAFTER WOOD hanya boleh membuat ECR kategori DRW (Wood) atau DBX (Box).'.
    GV_ICON    = 'error'.
    CONCATENATE 'drafter_list.htm?u=' GV_USER_ID '&r=' GV_ROLE '&n=' GV_FULL_NAME '&msg=' GV_MESSAGE '&icon=' GV_ICON INTO lv_url.
    navigation->goto_page( lv_url ). RETURN.
  ENDIF.
  IF GV_ROLE = 'DRAFTER_METAL' AND lv_kat <> 'DMT'.
    GV_MESSAGE = 'Role DRAFTER METAL hanya boleh membuat ECR kategori DMT (Metal).'.
    GV_ICON    = 'error'.
    CONCATENATE 'drafter_list.htm?u=' GV_USER_ID '&r=' GV_ROLE '&n=' GV_FULL_NAME '&msg=' GV_MESSAGE '&icon=' GV_ICON INTO lv_url.
    navigation->goto_page( lv_url ). RETURN.
  ENDIF.

  " Validate user session (prevent empty requester/drafter)
  IF GV_USER_ID IS INITIAL.
    GV_USER_ID = request->get_form_field( 'u' ).
  ENDIF.
  IF GV_USER_ID IS INITIAL.
    GV_MESSAGE = 'Session expired. Silakan login ulang.'. GV_ICON = 'error'.
    navigation->goto_page( 'login.htm' ). RETURN.
  ENDIF.

  " Validation
  IF lv_title IS INITIAL.
    GV_MESSAGE = 'Judul wajib diisi.'. GV_ICON = 'error'.
    CONCATENATE 'drafter_list.htm?u=' GV_USER_ID '&r=' GV_ROLE '&n=' GV_FULL_NAME '&msg=' GV_MESSAGE '&icon=' GV_ICON INTO lv_url.
    navigation->goto_page( lv_url ). RETURN.
  ENDIF.
  IF lv_matnr_str IS INITIAL.
    GV_MESSAGE = 'Pilih material.'. GV_ICON = 'error'.
    CONCATENATE 'drafter_list.htm?u=' GV_USER_ID '&r=' GV_ROLE '&n=' GV_FULL_NAME '&msg=' GV_MESSAGE '&icon=' GV_ICON INTO lv_url.
    navigation->goto_page( lv_url ). RETURN.
  ENDIF.
  IF lv_kat IS INITIAL OR lv_prio IS INITIAL OR lv_desc IS INITIAL OR lv_dept IS INITIAL OR lv_reason IS INITIAL.
    GV_MESSAGE = 'Semua field wajib diisi.'. GV_ICON = 'error'.
    CONCATENATE 'drafter_list.htm?u=' GV_USER_ID '&r=' GV_ROLE '&n=' GV_FULL_NAME '&msg=' GV_MESSAGE '&icon=' GV_ICON INTO lv_url.
    navigation->goto_page( lv_url ). RETURN.
  ENDIF.

  " Create ECR per material
  DATA: lt_mat TYPE STANDARD TABLE OF string,
        ls_mat TYPE string.
  SPLIT lv_matnr_str AT ',' INTO TABLE lt_mat.

  LOOP AT lt_mat INTO ls_mat.
    CONDENSE ls_mat NO-GAPS.
    CHECK ls_mat IS NOT INITIAL.

    CALL FUNCTION 'NUMBER_GET_NEXT'
      EXPORTING nr_range_nr = '01' object = 'ZECM_REQ'
      IMPORTING number = lv_ecr_no
      EXCEPTIONS OTHERS = 1.

    IF sy-subrc <> 0.
      GV_MESSAGE = 'Gagal generate nomor ECR.'. GV_ICON = 'error'.
      CONCATENATE 'drafter_list.htm?u=' GV_USER_ID '&r=' GV_ROLE '&n=' GV_FULL_NAME '&msg=' GV_MESSAGE '&icon=' GV_ICON INTO lv_url.
      navigation->goto_page( lv_url ). RETURN.
    ENDIF.

    SELECT SINGLE maktx FROM makt INTO @lv_maktx
      WHERE matnr = @ls_mat AND spras = @sy-langu.

    CLEAR ls_req.
    ls_req-mandt         = sy-mandt.
    ls_req-ecr_no        = lv_ecr_no.
    ls_req-matnr_fg      = ls_mat.
    ls_req-title         = lv_title.
    ls_req-descr         = lv_desc.
    ls_req-kategori      = lv_kat.
    ls_req-priority      = lv_prio.
    ls_req-requester     = GV_USER_ID.
    ls_req-drafter       = GV_USER_ID.
    ls_req-status        = 'CRTD'.
    ls_req-created_by    = GV_USER_ID.
    ls_req-created_date  = sy-datum.
    ls_req-created_time  = sy-uzeit.
    ls_req-change_scope  = 'DRWG'.
    ls_req-zdept         = lv_dept.
    ls_req-zreason       = lv_reason.
    IF lv_cost IS NOT INITIAL.
      ls_req-cost_estimate = lv_cost.
    ENDIF.

    INSERT zecm_req FROM ls_req.

    IF sy-subrc = 0.
      " Audit log
      CALL FUNCTION 'NUMBER_GET_NEXT'
        EXPORTING nr_range_nr = '01' object = 'ZECM_LOG'
        IMPORTING number = ls_log-log_id
        EXCEPTIONS OTHERS = 1.

      IF sy-subrc = 0.
        ls_log-mandt       = sy-mandt.
        ls_log-ecr_no      = lv_ecr_no.
        ls_log-action      = 'CREA'.
        ls_log-new_status  = 'CRTD'.
        ls_log-user_id     = GV_USER_ID.
        ls_log-action_date = sy-datum.
        ls_log-action_time = sy-uzeit.
        INSERT zecm_log FROM ls_log.
      ENDIF.

      lv_count = lv_count + 1.
    ENDIF.
  ENDLOOP.

  COMMIT WORK AND WAIT.

  DATA lv_msg TYPE string.
  lv_msg = lv_count.
  CONCATENATE lv_msg 'ECR berhasil dibuat' INTO GV_MESSAGE SEPARATED BY ' '.
  GV_ICON = 'success'.
  CONCATENATE 'drafter_list.htm?u=' GV_USER_ID '&r=' GV_ROLE '&n=' GV_FULL_NAME '&msg=' GV_MESSAGE '&icon=' GV_ICON INTO lv_url.
  navigation->goto_page( lv_url ).
  RETURN.
ENDIF.

" SUBMIT ECR (INWK -> INSP) + DMS (IA -> AP)
IF lv_action = 'submit_ecr'.
  lv_ecr_no = request->get_form_field( 'ecr_no' ).

  IF lv_ecr_no IS INITIAL.
    GV_MESSAGE = 'ECR tidak valid.'.
    GV_ICON    = 'error'.
    RETURN.
  ENDIF.

  " Cek status harus BMAP (BOM Approved) atau INWK jika dalam siklus revisi
  DATA: lv_cur_status    TYPE zecm_req-status,
        lv_cur_bom_check TYPE zecm_req-bom_check.
  SELECT SINGLE status, bom_check FROM zecm_req 
    INTO (@lv_cur_status, @lv_cur_bom_check)
    WHERE ecr_no = @lv_ecr_no.

  IF lv_cur_status <> 'BMAP' AND NOT ( lv_cur_status = 'INWK' AND ( lv_cur_bom_check = 'A' OR lv_cur_bom_check IS INITIAL ) ).
    GV_MESSAGE = 'BOM belum di-approve Master Data. Status harus BOM Approved.'.
    GV_ICON    = 'error'.
    RETURN.
  ENDIF.

  SELECT SINGLE matnr_fg FROM zecm_req INTO @lv_matnr_conv
    WHERE ecr_no = @lv_ecr_no.

  " DMS detection — Prioritas 1: dir_number tersimpan di ECR
  CLEAR: lv_doknr, lv_dokar, lv_dokvr, lv_doktl, lv_dokst.
  DATA: lv_cur_dir_sub TYPE zecm_req-dir_number.
  SELECT SINGLE dir_number FROM zecm_req INTO @lv_cur_dir_sub WHERE ecr_no = @lv_ecr_no.

  IF lv_cur_dir_sub IS NOT INITIAL.
    DATA: lt_sub_tokens TYPE TABLE OF string,
          lv_sub_cnt    TYPE i.
    SPLIT lv_cur_dir_sub AT '/' INTO TABLE lt_sub_tokens.
    DESCRIBE TABLE lt_sub_tokens LINES lv_sub_cnt.
    IF lv_sub_cnt >= 2.
      READ TABLE lt_sub_tokens INDEX 1 INTO lv_dokar.
      READ TABLE lt_sub_tokens INDEX 2 INTO lv_doknr.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
        EXPORTING  input  = lv_doknr

            SHIFT lv_ajax_clean_doknr LEFT DELETING LEADING '0'.
            CONCATENATE lv_ajax_dms_dokar '/' lv_ajax_clean_doknr INTO lv_ajax_saved_dir.

            UPDATE zecm_req SET
              dir_number   = @lv_ajax_saved_dir,
              changed_date = @sy-datum,
              changed_time = @sy-uzeit
            WHERE ecr_no = @lv_ajax_dms_ecr_no.

            COMMIT WORK AND WAIT.

            _m_response->set_content_type( 'application/json' ).
            DATA: lv_json_succ TYPE string.
            CONCATENATE '{"status":"success","message":"Berhasil","doknr":"' lv_ajax_ret_doknr '"}' INTO lv_json_succ.
            _m_response->set_cdata( lv_json_succ ).
            navigation->response_complete( ).
            RETURN.
          ELSE.
            _m_response->set_content_type( 'application/json' ).
            _m_response->set_cdata( '{"status":"error","message":"File tidak diproses oleh program SAP"}' ).
            navigation->response_complete( ).
            RETURN.
          ENDIF.
        ENDIF.
      ELSE.
        _m_response->set_content_type( 'application/json' ).
        _m_response->set_cdata( '{"status":"error","message":"Gagal menulis file ke direktori SAP"}' ).
        navigation->response_complete( ).
        RETURN.
      ENDIF.
    ENDIF.
    ENDIF.
    lv_ajax_idx = lv_ajax_idx + 1.
  ENDWHILE.
ENDIF.

  SELECT SINGLE matnr_fg, kategori FROM zecm_req

  DATA: lv_success_file_cnt TYPE i VALUE 0.
  
  lv_dms_numparts = request->num_multiparts( ).
  lv_dms_idx = 1.
  WHILE lv_dms_idx <= lv_dms_numparts.
    lo_dms_entity = request->get_multipart( lv_dms_idx ).
    IF lo_dms_entity->get_header_field( 'content-disposition' ) CS 'name="dms_file"'.

      lv_dms_filename = lo_dms_entity->get_header_field( '~content_filename' ).
      
      IF lv_dms_filebin IS NOT INITIAL AND lv_dms_filename IS NOT INITIAL.
        " Format nama file agar sesuai standard ZUPLOAD_CS_AUTO (Contoh: DRW_30013196_filename.dwg)
        DATA: lv_final_filename TYPE string,

        CONCATENATE lv_dms_dokar '_' lv_dms_matnr '_' lv_dms_filename INTO lv_final_filename.

        OPEN DATASET lv_al11_path FOR OUTPUT IN BINARY MODE.
        IF sy-subrc = 0.
          TRANSFER lv_dms_filebin TO lv_al11_path.
          CLOSE DATASET lv_al11_path.
          lv_success_file_cnt = lv_success_file_cnt + 1.
        ENDIF.
      ENDIF.
    ENDIF.
    lv_dms_idx = lv_dms_idx + 1.
  ENDWHILE.

  " Pengecekan
  IF lv_success_file_cnt = 0.
    GV_MESSAGE = 'File gagal ditangkap. Pastikan ukuran tidak melebihi batas server.'.

  " Jalankan ZECM_DMS_UPLOAD secara SYNCHRONOUS untuk memproses semua file yang sudah dikirim ke AL11
  SUBMIT zecm_dms_upload
    WITH p_dir = '/mnt/sap_media/engineering'
    WITH p_test = space
    WITH p_maxlv = 150
    WITH p_werks = '1000'
    WITH p_wer2 = '2000'
    EXPORTING LIST TO MEMORY
    AND RETURN.

           file   TYPE string,
           doknr  TYPE draw-doknr,

           msgv1  TYPE symsgv,
           msgv2  TYPE symsgv,
           msgv3  TYPE symsgv,
           msgv4  TYPE symsgv,
           msg    TYPE string,
         END OF ty_log.
  DATA: lt_upload_log TYPE STANDARD TABLE OF ty_log,
        ls_upload_log TYPE ty_log.

  IMPORT log = lt_upload_log FROM MEMORY ID 'ZECM_DMS_LOG'.

    GV_MESSAGE = 'File gagal diproses SAP (ZECM_DMS_UPLOAD gagal dieksekusi atau tidak mengirim respon memori).'.
    GV_ICON    = 'error'.
  ELSE.
    DATA: lv_err_cnt TYPE i VALUE 0,
          lv_suc_cnt TYPE i VALUE 0,
          lv_err_msg TYPE string.

    LOOP AT lt_upload_log INTO ls_upload_log.
      IF ls_upload_log-msgty = 'E' OR ls_upload_log-status = 'ERROR' OR ls_upload_log-status = 'SKIP'.
        lv_err_cnt = lv_err_cnt + 1.
        IF lv_err_msg IS INITIAL.
          lv_err_msg = ls_upload_log-msg.
        ENDIF.
      ELSE.
        lv_suc_cnt = lv_suc_cnt + 1.
      ENDIF.
    ENDLOOP.

    IF lv_err_cnt > 0.
      GV_MESSAGE = |Upload Selesai: { lv_suc_cnt } Sukses, { lv_err_cnt } Gagal ({ lv_err_msg })|.
      GV_ICON    = 'warning'.
